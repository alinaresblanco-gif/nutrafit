/* ============================================================
    CONTROL TOTAL NUTRAFIT - VERSIÓN INTEGRAL (UNIFICADA)
    VERSIÓN MEJORADA CON SOPORTE PARA DESPENSA DINÁMICO
   ============================================================ */

const SPREADSHEET_ID = "1ynyoanVl3CX-a9B-I45iy1BE6vhdJ0hrrRfipziuKz8";
const ADMIN_EMAIL = "alinares.blanco@gmail.com";
const ADMIN_KEY = "Nutrafit-Admin-2026";

/**
 * doGet: Gestiona la carga de páginas HTML y el envío de datos JSON
 */
function doGet(e) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);

    if (e.parameter.action === "admin_aprobar" && e.parameter.adminKey === ADMIN_KEY && e.parameter.email) {
      return aprobarUsuarioNutrafit(e.parameter.email);
    }

    const tabla = e.parameter.tabla || e.parameter.hoja;

    // --- 1. LÓGICA DE CARGA DE VISTAS HTML ---
    if (e.parameter.v) {
      return HtmlService.createTemplateFromFile(e.parameter.v)
          .evaluate()
          .addMetaTag('viewport', 'width=device-width, initial-scale=1')
          .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
    }

    // --- 2. LÓGICA DE LECTURA DE DATOS (JSON) ---
    if (!tabla) {
      return HtmlService.createTemplateFromFile('index')
          .evaluate()
          .addMetaTag('viewport', 'width=device-width, initial-scale=1')
          .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
    }

    const tz = Session.getScriptTimeZone();

    // CASO: ALIMENTOS
    if (tabla === "alimentos") {
      try {
        const sheetNames = ["alimentos", "Alimentos", "ALIMENTOS"];
        let sheet = null;
        for (let name of sheetNames) {
          sheet = ss.getSheetByName(name);
          if (sheet) break;
        }
        if (!sheet) {
          Logger.log("Error: No se encontró la hoja de alimentos");
          return finalizarJSON([]);
        }
        const data = sheet.getDataRange().getValues();
        if (data.length <= 1) return finalizarJSON([]);
        data.shift();
        const dataLimpia = data.filter(fila => fila[0] && fila[0].toString().trim() !== "");
        Logger.log("Alimentos cargados: " + dataLimpia.length);
        return finalizarJSON(dataLimpia);
      } catch (err) {
        Logger.log("Error en alimentos: " + err.toString());
        return finalizarJSON([]);
      }
    }

    // CASO: MENÚS SEMANALES
    else if (tabla === "menus_semanales" || tabla === "menu_semanal") {
      const usuarioId = normalizarEmail(e.parameter.usuario_id);
      if (!usuarioId) return finalizarJSON([]);

      const infoUsr = obtenerUsuarioNutrafit(ss, usuarioId);
      if (!infoUsr || !infoUsr.fila || estadoUsuarioNutrafit(infoUsr.fila) !== "activo") {
        return finalizarJSON([]);
      }

      const sheet = ss.getSheetByName("menus_semanales");
      if (!sheet) return finalizarJSON([]);
      const vals = sheet.getDataRange().getValues();
      if (vals.length <= 1) return finalizarJSON([]);

      const cabeceras = vals.shift();

      const resultado = vals
        .filter(fila => normalizarEmail(fila[0]) === usuarioId)
        .map(fila => {
          const obj = {};
          cabeceras.forEach((cab, i) => {
            const celda = fila[i];
            if (celda instanceof Date) {
              obj[cab] = Utilities.formatDate(celda, tz, "yyyy-MM-dd");
            } else {
              obj[cab] = (celda !== null && celda !== undefined) ? celda : "";
            }
          });
          return obj;
        });

      return finalizarJSON(resultado);
    }

    // CASO: RECETAS
    else if (tabla === "recetas") {
      const sheetRec = ss.getSheetByName("recetas");
      if (!sheetRec) return finalizarJSON([]);
      const datos = sheetRec.getDataRange().getValues();
      if (datos.length <= 1) return finalizarJSON([]);
      datos.shift();
      const filasLimpias = datos.map(fila => fila.map(celda => {
        if (celda === null || celda === undefined) return "";
        if (celda instanceof Date) return celda.toLocaleDateString();
        return celda.toString();
      }));
      return finalizarJSON(filasLimpias);
    }

    // CASO: EJERCICIO
    else if (tabla === "ejercicio") {
      const usuarioId = normalizarEmail(e.parameter.usuario_id);
      if (!usuarioId) return finalizarJSON([]);

      const infoUsr = obtenerUsuarioNutrafit(ss, usuarioId);
      if (!infoUsr || !infoUsr.fila || estadoUsuarioNutrafit(infoUsr.fila) !== "activo") {
        return finalizarJSON([]);
      }

      const sheet = ss.getSheetByName("ejercicio");
      if (!sheet) return finalizarJSON([]);
      const data = sheet.getDataRange().getValues();
      if (data.length <= 1) return finalizarJSON([]);
      data.shift();

      // Columnas: [usuario_id, fecha, actividad, tiempo, foto, distancia, pasos, desnivel, vel_media]
      // Añadimos rowId al final para permitir borrado exacto desde frontend.
      const dataUsuario = data
        .map((fila, idx) => ({ fila, rowId: idx + 2 }))
        .filter(item => normalizarEmail(item.fila[0]) === usuarioId)
        .map(item => {
          const filaConvertida = item.fila.map(celda =>
            celda instanceof Date
              ? Utilities.formatDate(celda, tz, "yyyy-MM-dd'T'HH:mm:ss")
              : celda
          );
          filaConvertida.push(item.rowId);
          return filaConvertida;
        });
      return finalizarJSON(dataUsuario.slice(-20).reverse());
    }

    // CASO: PESO
    else if (tabla === "peso") {
      const usuarioId = normalizarEmail(e.parameter.usuario_id);
      if (!usuarioId) return finalizarJSON([]);

      const infoUsr = obtenerUsuarioNutrafit(ss, usuarioId);
      if (!infoUsr || !infoUsr.fila || estadoUsuarioNutrafit(infoUsr.fila) !== "activo") {
        return finalizarJSON([]);
      }

      const sheet = ss.getSheetByName("Peso");
      if (!sheet) return finalizarJSON([]);
      const data = sheet.getDataRange().getValues();
      if (data.length <= 1) return finalizarJSON([]);
      data.shift();

      // Añadimos rowId al final para permitir borrado exacto desde frontend.
      const dataUsuario = data
        .map((fila, idx) => ({ fila, rowId: idx + 2 }))
        .filter(item => normalizarEmail(item.fila[0]) === usuarioId)
        .map(item => {
          const filaConvertida = item.fila.map(celda =>
            celda instanceof Date ? Utilities.formatDate(celda, tz, "yyyy-MM-dd") : celda
          );
          filaConvertida.push(item.rowId);
          return filaConvertida;
        });
      return finalizarJSON(dataUsuario.slice(-15).reverse());
    }

    // CASO: CRÉDITOS
    else if (tabla === "creditos" || tabla === "CREDITOS") {
      const usuarioId = normalizarEmail(e.parameter.usuario_id);
      if (!usuarioId) return finalizarJSON([]);

      // Verificar usuario activo (igual que agua)
      const infoUsr = obtenerUsuarioNutrafit(ss, usuarioId);
      if (!infoUsr || !infoUsr.fila || estadoUsuarioNutrafit(infoUsr.fila) !== "activo") {
        return finalizarJSON([]);
      }

      const sheet = ss.getSheetByName("Creditos") || ss.getSheetByName("CREDITOS");
      if (!sheet) return finalizarJSON([]);
      const data = sheet.getDataRange().getValues();
      if (data.length <= 1) return finalizarJSON([]);
      data.shift();

      // Filtrar por usuario y convertir fechas
      const dataUsuario = data
        .filter(fila => normalizarEmail(fila[0]) === usuarioId)
        .map(fila => fila.map(celda =>
          celda instanceof Date ? Utilities.formatDate(celda, tz, "yyyy-MM-dd") : celda
        ));
      return finalizarJSON(dataUsuario.slice(-10).reverse());
    }

    // CASO: AGUA
    else if (tabla === "agua") {
      const usuarioId = normalizarEmail(e.parameter.usuario_id);

      // Para lectura solo se exige que el usuario exista y esté activo.
      // El control de dispositivo se aplica únicamente en escritura (doPost).
      if (!usuarioId) return finalizarJSON([]);
      const infoUsuario = obtenerUsuarioNutrafit(ss, usuarioId);
      if (!infoUsuario || !infoUsuario.fila || estadoUsuarioNutrafit(infoUsuario.fila) !== "activo") {
        return finalizarJSON([]);
      }

      const sheet = ss.getSheetByName("Control_Agua");
      if (!sheet) return finalizarJSON([]);
      const data = sheet.getDataRange().getValues();
      if (data.length <= 1) return finalizarJSON([]);
      data.shift();
      const dataUsuario = data.filter(fila => {
        // Nuevo formato: [usuario_id, fecha, vasos, estado, hora]
        if (Array.isArray(fila) && fila.length >= 5) {
          return normalizarEmail(fila[0]) === usuarioId;
        }

        // Formato legado: [fecha, vasos, estado, hora]
        // No tenía usuario_id, así que se mantiene compatibilidad para no perder historial previo.
        return Array.isArray(fila) && fila.length >= 4;
      });
      return finalizarJSON(dataUsuario.slice(-10).reverse());
    }

    else {
      return finalizarJSON([]);
    }

  } catch (err) {
    Logger.log("Error en doGet: " + err.toString());
    return ContentService.createTextOutput(JSON.stringify({ error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * doPost: Gestión de guardado de datos
 */
function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var tipo = payload.tipo || payload.accion || payload.action;

    if (tipo === "verificar_acceso") {
      return verificarAccesoNutrafit(ss, payload);
    }

    if (tipo === "solicitar_acceso") {
      return solicitarAccesoNutrafit(ss, payload);
    }

    if (tipo === "coach_nutrafit") {
      return procesarCoachNutrafit(ss, payload);
    }


    // REGISTRO DE USUARIO
    if (tipo === "registro_usuario") {
      const hojaUsuarios = ss.getSheetByName("usuarios");
      const { usuario_id, nombre, dispositivo_id } = payload;
      const datos = hojaUsuarios.getDataRange().getValues();
      const existe = datos.findIndex(fila => fila[0] === usuario_id);
      if (existe !== -1) {
        return ContentService.createTextOutput(JSON.stringify({ status: "existe" })).setMimeType(ContentService.MimeType.JSON);
      }
      const fecha_registro = new Date();
      hojaUsuarios.appendRow([
        usuario_id,
        nombre,
        Utilities.formatDate(fecha_registro, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"),
        false,
        dispositivo_id,
        "",
        false
      ]);
      // Notificar al admin
      MailApp.sendEmail({
        to: "alinares.blanco@gmail.com", // <-- CAMBIA POR TU CORREO
        subject: "Nuevo registro en Nutrafit",
        body: `Nuevo usuario registrado: ${nombre} (${usuario_id})\n\nValídalo en la hoja de usuarios.`
      });
      return ContentService.createTextOutput(JSON.stringify({ status: "pendiente" })).setMimeType(ContentService.MimeType.JSON);
    }

    // LOGIN DE USUARIO
    if (tipo === "login_usuario") {
      const hojaUsuarios = ss.getSheetByName("usuarios");
      const { usuario_id, dispositivo_id } = payload;
      const datos = hojaUsuarios.getDataRange().getValues();
      const fila = datos.find(fila => fila[0] === usuario_id);
      if (!fila) {
        return ContentService.createTextOutput(JSON.stringify({ status: "no_registrado" })).setMimeType(ContentService.MimeType.JSON);
      }
      if (!fila[3]) {
        return ContentService.createTextOutput(JSON.stringify({ status: "no_validado" })).setMimeType(ContentService.MimeType.JSON);
      }
      if (fila[4] !== dispositivo_id) {
        return ContentService.createTextOutput(JSON.stringify({ status: "otro_dispositivo" })).setMimeType(ContentService.MimeType.JSON);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "ok", nombre: fila[1] })).setMimeType(ContentService.MimeType.JSON);
    }

    // NOTIFICAR VALIDACIÓN DE USUARIO (ADMIN)
    if (tipo === "notificar_validacion") {
      const hojaUsuarios = ss.getSheetByName("usuarios");
      const { usuario_id } = payload;
      const datos = hojaUsuarios.getDataRange().getValues();
      const idx = datos.findIndex(fila => fila[0] === usuario_id);
      if (idx === -1) {
        return ContentService.createTextOutput("Usuario no encontrado");
      }
      const fila = datos[idx];
      if (fila[3] && !fila[5]) {
        // Solo si está validado y no se ha notificado
        const fecha_validacion = new Date();
        hojaUsuarios.getRange(idx + 1, 6).setValue(Utilities.formatDate(fecha_validacion, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"));
        // Notificar al usuario
        MailApp.sendEmail({
          to: fila[0],
          subject: "Nutrafit: Usuario validado",
          body: `Hola ${fila[1]},\n\nTu usuario ha sido validado. Ya puedes acceder a Nutrafit.\n\n¡Bienvenida/o!`
        });
        return ContentService.createTextOutput("Notificado");
      }
      return ContentService.createTextOutput("No validado o ya notificado");
    }

    // GUARDAR EJERCICIO
    if (tipo === "guardar_ejercicio") {
      const sheetEj = ss.getSheetByName("ejercicio");
      const usuarioId = normalizarEmail(payload.usuario_id);
      const dispositivoId = String(payload.dispositivo_id || "").trim();

      if (!sheetEj) {
        return ContentService.createTextOutput("No existe la hoja ejercicio");
      }

      if (!usuarioId || !dispositivoId || !sesionValidaNutrafit(ss, usuarioId, dispositivoId)) {
        return ContentService.createTextOutput("Sesión inválida");
      }

      var celdaFoto = (payload.imagenBase64 && payload.imagenBase64.length > 100) ? "data:image/jpeg;base64," + payload.imagenBase64 : "";
      var distLim = parseFloat(String(payload.distancia || "0").replace("'", "").replace(",", "."));
      var tiempoLim = parseFloat(String(payload.tiempo || "0").replace("'", "").replace(",", "."));
      var velMedia = (distLim > 0 && tiempoLim > 0) ? (distLim / (tiempoLim / 60)).toFixed(2) : 0;

      sheetEj.appendRow([usuarioId, new Date(), payload.actividad, tiempoLim, celdaFoto, distLim, payload.pasos || 0, payload.desnivel || 0, velMedia]);
      return ContentService.createTextOutput("Éxito Ejercicio");
    }

    // GUARDAR RECETA
    else if (tipo === "guardar_receta") {
      var sheetRec = ss.getSheetByName("recetas") || ss.insertSheet("recetas");
      var celdaFotoRec = (payload.imagenBase64 && payload.imagenBase64.length > 100) ? "data:image/jpeg;base64," + payload.imagenBase64 : "";
      sheetRec.appendRow([payload.usuario_id || "comun", new Date(), celdaFotoRec, payload.nombre, payload.categoria, payload.ingredientes, payload.elaboracion]);
      return ContentService.createTextOutput("Éxito Receta");
    }

    // GUARDAR CRÉDITOS
    else if (tipo === "creditos") {
      var sheetCreditos = ss.getSheetByName("Creditos") || ss.getSheetByName("CREDITOS");
      sheetCreditos.appendRow([payload.usuario_id, payload.fecha, payload.genero, payload.edad, payload.peso, payload.altura, payload.total]);
      return ContentService.createTextOutput("Éxito Créditos");
    }

    // GUARDAR MENÚ SEMANAL
    else if (tipo === "guardar_menu_semanal" || tipo === "guardarFila" || tipo === "actualizar_menu_semanal") {
      var sheetMenus = ss.getSheetByName("menus_semanales") || ss.insertSheet("menus_semanales");
      const usuarioId = normalizarEmail(payload.usuario_id);
      const dispositivoId = String(payload.dispositivo_id || "").trim();
      const esActualizacion = (tipo === "actualizar_menu_semanal") || (String(payload.modo || "").toLowerCase() === "actualizar");

      if (!usuarioId || !dispositivoId || !sesionValidaNutrafit(ss, usuarioId, dispositivoId)) {
        return ContentService.createTextOutput("Sesión inválida");
      }

      // ✅ Añadimos usuario_id como primera columna
      const headers = ["usuario_id", "Semana inicio", "Fecha", "Día", "Momento", "Ingrediente", "Cantidad", "Puntos", "Notas", "Presupuesto"];

      if (sheetMenus.getLastRow() === 0) {
        sheetMenus.appendRow(headers);
      } else {
        const filaCabecera = sheetMenus.getRange(1, 1, 1, headers.length).getValues()[0];
        if (filaCabecera.join("|") !== headers.join("|")) {
          sheetMenus.getRange(1, 1, 1, headers.length).setValues([headers]);
        }
      }

      const filas = Array.isArray(payload.filas)
        ? payload.filas
        : Array.isArray(payload.datos)
          ? payload.datos
          : [payload.datos || payload];

      const filasParaGuardar = filas
        .filter(f => f && typeof f === 'object' && Object.keys(f).length > 0)
        .map(f => [
          usuarioId,
          sanitizeCampo(f["Semana inicio"] || f.semana_inicio || f.semana || ""),
          sanitizeCampo(f["Fecha"]         || f.fecha         || ""),
          sanitizeCampo(f["Día"]           || f.Dia           || f.dia || ""),
          sanitizeCampo(f["Momento"]       || f.momento       || ""),
          sanitizeCampo(f["Ingrediente"]   || f.ingrediente   || ""),
          "",
          sanitizeCampo(f["Puntos"]        || f.puntos        || ""),
          sanitizeCampo(f["Notas"]         || f.notas         || ""),
          sanitizeCampo(f["Presupuesto"]   || f.presupuesto   || payload.presupuesto || "")
        ]);

      // Evita duplicados en el propio payload antes de tocar la hoja.
      const filasUnicasMap = {};
      filasParaGuardar.forEach(fila => {
        const clave = [fila[1], fila[2], fila[3], fila[4], fila[5]]
          .map(valor => String(valor || "").trim().toLowerCase())
          .join("|");
        filasUnicasMap[clave] = fila;
      });
      const filasParaGuardarUnicas = Object.keys(filasUnicasMap).map(clave => filasUnicasMap[clave]);

      function claveFilaMenu(fila) {
        // Clave lógica de una comida dentro de la semana (ignora mayúsculas/minúsculas).
        return [fila[1], fila[2], fila[3], fila[4], fila[5]]
          .map(valor => String(valor || "").trim().toLowerCase())
          .join("|");
      }

      // Semanas incluidas en esta operación.
      const semanasAReemplazar = Array.from(new Set(
        filasParaGuardarUnicas
          .map(fila => String(fila[1] || '').trim()) // Columna "Semana inicio"
          .filter(valor => valor !== '')
      ));

      if (semanasAReemplazar.length > 0 && sheetMenus.getLastRow() > 1) {
        const dataActual = sheetMenus.getDataRange().getValues();

        if (esActualizacion) {
          // Actualizar semana activa: reemplazo total por usuario+semana para evitar duplicados acumulados.
          const filasAEliminar = [];
          dataActual.slice(1).forEach((fila, idx) => {
            const filaUsuario = normalizarEmail(fila[0]);
            const filaSemanaInicio = String(fila[1] || '').trim();
            if (filaUsuario === usuarioId && semanasAReemplazar.includes(filaSemanaInicio)) {
              filasAEliminar.push(idx + 2); // row number real en la hoja
            }
          });

          filasAEliminar.sort((a, b) => b - a).forEach(rowNum => sheetMenus.deleteRow(rowNum));

          if (filasParaGuardarUnicas.length > 0) {
            sheetMenus.getRange(
              sheetMenus.getLastRow() + 1, 1,
              filasParaGuardarUnicas.length, headers.length
            ).setValues(filasParaGuardarUnicas);
          }
        } else {
          // Guardar semana nueva: no sobrescribe una semana existente.
          const existeSemana = dataActual.slice(1).some(fila => {
            const filaUsuario = normalizarEmail(fila[0]);
            const filaSemanaInicio = String(fila[1] || '').trim();
            return filaUsuario === usuarioId && semanasAReemplazar.includes(filaSemanaInicio);
          });

          if (existeSemana) {
            return ContentService
              .createTextOutput("Ya existe una semana con esa fecha. Usa Actualizar para modificarla.")
              .setMimeType(ContentService.MimeType.TEXT);
          }

          if (filasParaGuardarUnicas.length > 0) {
            sheetMenus.getRange(
              sheetMenus.getLastRow() + 1, 1,
              filasParaGuardarUnicas.length, headers.length
            ).setValues(filasParaGuardarUnicas);
          }
        }
      } else if (filasParaGuardarUnicas.length > 0) {
        // Primera carga (sin semanas previas) o sin colisiones.
        sheetMenus.getRange(
          sheetMenus.getLastRow() + 1, 1,
          filasParaGuardarUnicas.length, headers.length
        ).setValues(filasParaGuardarUnicas);
      }

      return ContentService.createTextOutput(esActualizacion ? "Éxito Actualizar Menú Semanal" : "Éxito Menú Semanal");
    }

    // GUARDAR PESO
    else if (tipo === "peso") {
      const sheetPeso = ss.getSheetByName("Peso");
      sheetPeso.appendRow([payload.usuario_id, payload.fecha, payload.peso, payload.diferencia]);
      return ContentService.createTextOutput("Éxito Peso");
    }

    // ELIMINAR REGISTRO DE PESO
    else if (tipo === "eliminar_peso") {
      const sheetPeso = ss.getSheetByName("Peso");
      const usuarioId = normalizarEmail(payload.usuario_id);
      const dispositivoId = String(payload.dispositivo_id || "").trim();
      const rowId = Number(payload.row_id || 0);
      const fecha = String(payload.fecha || "").trim();
      const peso = String(payload.peso || "").trim();

      if (!sheetPeso) {
        return ContentService.createTextOutput("No existe la hoja Peso");
      }

      if (!usuarioId || !dispositivoId || !sesionValidaNutrafit(ss, usuarioId, dispositivoId)) {
        return ContentService.createTextOutput("Sesión inválida");
      }

      // 1) Intento principal: borrar por row_id exacto (viene de doGet peso)
      if (rowId > 1 && rowId <= sheetPeso.getLastRow()) {
        const fila = sheetPeso.getRange(rowId, 1, 1, 4).getValues()[0];
        if (normalizarEmail(fila[0]) === usuarioId) {
          sheetPeso.deleteRow(rowId);
          return ContentService.createTextOutput("Éxito Eliminar Peso");
        }
      }

      // 2) Fallback: buscar por usuario + fecha + peso desde abajo (más reciente primero)
      const data = sheetPeso.getDataRange().getValues();
      for (let i = data.length - 1; i >= 1; i--) {
        const fila = data[i];
        const filaUsuario = normalizarEmail(fila[0]);
        const filaFecha = String(fila[1] || "").trim();
        const filaPeso = String(fila[2] || "").trim();

        if (filaUsuario === usuarioId && filaFecha === fecha && filaPeso === peso) {
          sheetPeso.deleteRow(i + 1);
          return ContentService.createTextOutput("Éxito Eliminar Peso");
        }
      }

      return ContentService.createTextOutput("Registro no encontrado");
    }

    // ELIMINAR REGISTRO DE EJERCICIO
    else if (tipo === "eliminar_ejercicio") {
      const sheetEj = ss.getSheetByName("ejercicio");
      const usuarioId = normalizarEmail(payload.usuario_id);
      const dispositivoId = String(payload.dispositivo_id || "").trim();
      const rowId = Number(payload.row_id || 0);
      const fechaVal = String(payload.fecha || "").trim();
      const actividad = String(payload.actividad || "").trim();

      if (!sheetEj) {
        return ContentService.createTextOutput("No existe la hoja ejercicio");
      }

      if (!usuarioId || !dispositivoId || !sesionValidaNutrafit(ss, usuarioId, dispositivoId)) {
        return ContentService.createTextOutput("Sesión inválida");
      }

      // 1) Intento principal: borrar por row_id exacto
      if (rowId > 1 && rowId <= sheetEj.getLastRow()) {
        const fila = sheetEj.getRange(rowId, 1, 1, 3).getValues()[0];
        if (normalizarEmail(fila[0]) === usuarioId) {
          sheetEj.deleteRow(rowId);
          return ContentService.createTextOutput("Éxito Eliminar Ejercicio");
        }
      }

      // 2) Fallback: buscar por usuario + fecha + actividad desde abajo
      const data = sheetEj.getDataRange().getValues();
      for (let i = data.length - 1; i >= 1; i--) {
        const fila = data[i];
        const filaUsuario = normalizarEmail(fila[0]);
        const filaFecha = String(fila[1] instanceof Date
          ? Utilities.formatDate(fila[1], Session.getScriptTimeZone(), "yyyy-MM-dd'T'HH:mm:ss")
          : fila[1] || "").trim();
        const filaActividad = String(fila[2] || "").trim();
        if (filaUsuario === usuarioId && filaFecha.startsWith(fechaVal.substring(0, 10)) && filaActividad === actividad) {
          sheetEj.deleteRow(i + 1);
          return ContentService.createTextOutput("Éxito Eliminar Ejercicio");
        }
      }

      return ContentService.createTextOutput("Registro no encontrado");
    }

    // GUARDAR ALIMENTO NUEVO
    else if (tipo === "guardar_alimento") {
      const sheetAlim = ss.getSheetByName("Alimentos") || ss.getSheetByName("alimentos") || ss.insertSheet("Alimentos");
      sheetAlim.appendRow([payload.nombre, payload.grupo, payload.proteinas, payload.carbohidratos, payload.grasas, payload.fibra, payload.manual, payload.calculado, payload.neto, new Date()]);
      return ContentService.createTextOutput("Éxito Alimento");
    }

    // GUARDAR AGUA (POR DEFECTO)
    else {
      const sheetAgua = ss.getSheetByName("Control_Agua");
      const usuarioId = normalizarEmail(payload.usuario_id);
      const dispositivoId = String(payload.dispositivo_id || "").trim();

      if (!sheetAgua) {
        return ContentService.createTextOutput("No existe la hoja Control_Agua");
      }

      if (!usuarioId || !dispositivoId || !sesionValidaNutrafit(ss, usuarioId, dispositivoId)) {
        return ContentService.createTextOutput("Sesión inválida");
      }

      const fechaActual = new Date();
      sheetAgua.appendRow([
        usuarioId,
        fechaActual.toLocaleDateString(),
        (payload.vasos || 0) + " vasos",
        (payload.vasos >= 8) ? "COMPLETADO" : "PENDIENTE",
        fechaActual.toLocaleTimeString()
      ]);
      return ContentService.createTextOutput("Éxito Agua");
    }

  } catch (err) {
    Logger.log("Error en doPost: " + err.toString());
    return ContentService.createTextOutput(JSON.stringify({ error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function obtenerAlimentosDespensa() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const hoja = ss.getSheetByName("alimentos") || ss.getSheetByName("Alimentos");
    if (!hoja) return [];
    const datos = hoja.getDataRange().getValues();
    const cabeceras = datos.shift();
    const idxNombre = cabeceras.indexOf("Nombre");
    const idxNetos = cabeceras.indexOf("Netos");
    if (idxNombre === -1 || idxNetos === -1) return [];
    return datos
      .map(fila => ({ nombre: fila[idxNombre], netos: fila[idxNetos] }))
      .filter(item => item.nombre !== "" && item.nombre !== undefined);
  } catch (e) {
    Logger.log("Error obtenerAlimentosDespensa: " + e.toString());
    return [];
  }
}

function asegurarCabeceraUsuarios(hojaUsuarios) {
  const cabeceraEsperada = ["usuario_id", "nombre", "fecha_registro", "validado", "dispositivo_id", "fecha_validacion", "admin"];
  const ultimaFila = hojaUsuarios.getLastRow();

  if (ultimaFila === 0) {
    hojaUsuarios.appendRow(cabeceraEsperada);
    return;
  }

  const cabeceraActual = hojaUsuarios.getRange(1, 1, 1, cabeceraEsperada.length).getValues()[0];
  const normalizada = cabeceraActual.map(valor => String(valor || "").trim().toLowerCase());
  const esperada = cabeceraEsperada.map(valor => String(valor || "").trim().toLowerCase());

  if (normalizada.join("|") !== esperada.join("|")) {
    hojaUsuarios.getRange(1, 1, 1, cabeceraEsperada.length).setValues([cabeceraEsperada]);
  }
}

function normalizarEmail(valor) {
  return String(valor || "").trim().toLowerCase();
}

function obtenerUsuarioNutrafit(ss, email) {
  const hojaUsuarios = ss.getSheetByName("usuarios");
  if (!hojaUsuarios) return null;

  asegurarCabeceraUsuarios(hojaUsuarios);

  const datos = hojaUsuarios.getDataRange().getValues();
  const emailBuscado = normalizarEmail(email);

  for (let i = 1; i < datos.length; i++) {
    const fila = datos[i];
    if (normalizarEmail(fila[0]) === emailBuscado) {
      return { hojaUsuarios, datos, fila, indice: i };
    }
  }

  return { hojaUsuarios, datos, fila: null, indice: -1 };
}

function sesionValidaNutrafit(ss, usuarioId, dispositivoId) {
  const usuario = normalizarEmail(usuarioId);
  const dispositivo = String(dispositivoId || "").trim();
  if (!usuario || !dispositivo) return false;

  const info = obtenerUsuarioNutrafit(ss, usuario);
  if (!info || !info.fila) return false;

  const estado = estadoUsuarioNutrafit(info.fila);
  if (estado !== "activo") return false;

  const dispositivoGuardado = String(info.fila[4] || "").trim();

  // Primera asociación de dispositivo para usuarios activos sin dispositivo registrado.
  if (!dispositivoGuardado) {
    info.hojaUsuarios.getRange(info.indice + 1, 5).setValue(dispositivo);
    return true;
  }

  return dispositivoGuardado === dispositivo;
}

function estadoUsuarioNutrafit(fila) {
  if (!fila) return "no_existe";
  const estadoColumna = fila[3];
  if (estadoColumna === true || estadoColumna === "true") return "activo";
  if (String(estadoColumna).toLowerCase() === "bloqueado") return "bloqueado";
  if (String(estadoColumna).toLowerCase() === "activo") return "activo";
  if (String(fila[0] || "").trim() === "") return "no_existe";
  return "pendiente";
}

function verificarAccesoNutrafit(ss, payload) {
  const email = normalizarEmail(payload.email || payload.usuario_id);
  const dispositivoId = String(payload.dispositivo_id || "").trim();
  if (!email) {
    return finalizarJSON({ status: "error", mensaje: "Email requerido" });
  }

  const info = obtenerUsuarioNutrafit(ss, email);
  if (!info || !info.fila) {
    return finalizarJSON({ status: "no_existe" });
  }

  const estado = estadoUsuarioNutrafit(info.fila);
  if (estado === "activo") {
    // Sincroniza el dispositivo activo para que el guardado posterior de datos no falle por sesión.
    if (dispositivoId) {
      const dispositivoGuardado = String(info.fila[4] || "").trim();
      if (dispositivoGuardado !== dispositivoId) {
        info.hojaUsuarios.getRange(info.indice + 1, 5).setValue(dispositivoId);
      }
    }

    info.hojaUsuarios.getRange(info.indice + 1, 6).setValue(Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"));
    return finalizarJSON({
      status: "activo",
      usuario: {
        email: info.fila[0],
        nombre: info.fila[1] || ""
      }
    });
  }

  if (estado === "bloqueado") {
    return finalizarJSON({ status: "bloqueado" });
  }

  return finalizarJSON({ status: "pendiente" });
}

function solicitarAccesoNutrafit(ss, payload) {
  const email = normalizarEmail(payload.email || payload.usuario_id);
  const nombre = String(payload.nombre || "").trim();

  if (!email || !nombre) {
    return finalizarJSON({ status: "error", mensaje: "Email y nombre son obligatorios" });
  }

  const info = obtenerUsuarioNutrafit(ss, email);
  if (!info || !info.hojaUsuarios) {
    return finalizarJSON({ status: "error", mensaje: "No existe la hoja de usuarios" });
  }

  const estado = info.fila ? estadoUsuarioNutrafit(info.fila) : "no_existe";
  if (estado === "activo") {
    return finalizarJSON({ status: "activo" });
  }

  if (estado === "pendiente") {
    enviarCorreoSolicitudAdminNutrafit(email, nombre, new Date().toISOString(), ss, info.hojaUsuarios, info.indice);
    return finalizarJSON({ status: "pendiente", reenviado: true });
  }

  if (info.fila) {
    return finalizarJSON({ status: estado });
  }

  const fecha_registro = new Date();
  info.hojaUsuarios.appendRow([
    email,
    nombre,
    Utilities.formatDate(fecha_registro, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"),
    false,
    "",
    "",
    false
  ]);

  enviarCorreoSolicitudAdminNutrafit(email, nombre, Utilities.formatDate(fecha_registro, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"), ss);

  return finalizarJSON({ status: "pendiente" });
}

function enviarCorreoSolicitudAdminNutrafit(email, nombre, fechaIso) {
  const urlBase = ScriptApp.getService().getUrl();
  const urlAprobar = urlBase
    ? urlBase + "?action=admin_aprobar&email=" + encodeURIComponent(email) + "&adminKey=" + encodeURIComponent(ADMIN_KEY)
    : "";

  MailApp.sendEmail({
    to: ADMIN_EMAIL,
    subject: "Nuevo registro en Nutrafit: " + nombre,
    body:
      "Nueva solicitud de acceso\n" +
      "Nombre: " + nombre + "\n" +
      "Email: " + email + "\n" +
      "Fecha: " + fechaIso + "\n\n" +
      (urlAprobar ? ("Aprobar: " + urlAprobar + "\n") : "")
  });
}

function aprobarUsuarioNutrafit(email) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const info = obtenerUsuarioNutrafit(ss, email);
  if (!info || !info.fila) {
    return textoPlano("Usuario no encontrado");
  }

  info.hojaUsuarios.getRange(info.indice + 1, 4).setValue(true);
  info.hojaUsuarios.getRange(info.indice + 1, 6).setValue(Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"));
  info.hojaUsuarios.getRange(info.indice + 1, 7).setValue(ADMIN_EMAIL);

  MailApp.sendEmail({
    to: normalizarEmail(email),
    subject: "Nutrafit: acceso aprobado",
    body: "Hola " + (info.fila[1] || "") + ",\n\nTu acceso ha sido aprobado. Ya puedes entrar en Nutrafit."
  });

  return textoPlano("Usuario aprobado y notificado");
}

function procesarCoachNutrafit(ss, payload) {
  const usuarioId = normalizarEmail(payload.usuario_id);
  const dispositivoId = String(payload.dispositivo_id || "").trim();
  const evento = String(payload.evento || "pregunta_libre").trim();
  const modo = String(payload.modo || "auto").toLowerCase() === "chat" ? "chat" : "auto";
  const pregunta = String(payload.pregunta || "").trim();
  const contexto = payload.contexto && typeof payload.contexto === "object" ? payload.contexto : {};

  if (!coachHabilitadoGlobal()) {
    return finalizarJSON({ status: "disabled", message: "Coach pausado globalmente" });
  }

  if (!usuarioId || !dispositivoId || !sesionValidaNutrafit(ss, usuarioId, dispositivoId)) {
    return finalizarJSON({ status: "error", message: "Sesion invalida" });
  }

  const cfg = obtenerConfigCoach();
  const cuota = validarCuotaCoachDiaria(ss, usuarioId, modo, cfg);
  if (!cuota.ok) {
    return finalizarJSON({
      status: "quota_exceeded",
      message: "Limite diario del piloto alcanzado",
      quota: cuota
    });
  }

  const modelo = elegirModeloCoach(evento, payload, cfg);
  const accionNutrafit = mapearEventoAAccionNutrafit(evento, payload);
  const memoriaUsuario = obtenerMemoriaCoachUsuario(ss, usuarioId, 8);
  const contextoUsuario = construirContextoUsuarioCoach(ss, usuarioId);
  const contextoFinal = Object.assign({}, contexto || {}, {
    perfil_usuario: contextoUsuario,
    accion_nutrafit: accionNutrafit,
    memoria_usuario: memoriaUsuario
  });
  const mensajeUsuario = construirMensajeCoach(pregunta, evento, contextoFinal, modo);

  try {
    const respuestaGroq = solicitarGroqCoach(modelo, mensajeUsuario, cfg.maxTokensSalida, modo, evento);
    const contenido = String(
      (((respuestaGroq || {}).choices || [])[0] || {}).message
        ? (((respuestaGroq || {}).choices || [])[0] || {}).message.content || ""
        : ""
    ).trim();

    let respuesta = parsearRespuestaCoachJSON(contenido);
    respuesta = normalizarRespuestaCoach(respuesta, evento, contextoFinal, pregunta, contenido, modo);

    registrarMemoriaCoachUsuario(ss, {
      usuarioId: usuarioId,
      pregunta: pregunta,
      respuestaDirecta: String(respuesta.respuesta_directa || "").substring(0, 300),
      accion: accionNutrafit,
      intencion: detectarIntencionCoach(pregunta, evento)
    });

    const usage = respuestaGroq.usage || {};
    registrarUsoCoach(ss, {
      usuarioId,
      modo,
      evento,
      modelo,
      promptTokens: Number(usage.prompt_tokens || 0),
      completionTokens: Number(usage.completion_tokens || 0),
      totalTokens: Number(usage.total_tokens || 0),
      pregunta
    });

    return finalizarJSON({
      status: "ok",
      model: modelo,
      respuesta,
      quota: {
        used: cuota.used + 1,
        max: cuota.max,
        mode: modo
      }
    });
  } catch (error) {
    Logger.log("Error coach_nutrafit: " + error);
    const fallback = normalizarRespuestaCoach({}, evento, contextoFinal, pregunta, "", modo);

    registrarMemoriaCoachUsuario(ss, {
      usuarioId: usuarioId,
      pregunta: pregunta,
      respuestaDirecta: String(fallback.respuesta_directa || "").substring(0, 300),
      accion: accionNutrafit,
      intencion: detectarIntencionCoach(pregunta, evento)
    });

    registrarUsoCoach(ss, {
      usuarioId,
      modo,
      evento,
      modelo,
      promptTokens: 0,
      completionTokens: 0,
      totalTokens: 0,
      pregunta
    });

    return finalizarJSON({
      status: "ok",
      model: modelo,
      respuesta: fallback,
      warning: "fallback"
    });
  }
}

function coachHabilitadoGlobal() {
  const props = PropertiesService.getScriptProperties();
  const valor = String(props.getProperty("NUTRAFIT_COACH_ENABLED") || "true").toLowerCase();
  return !(valor === "false" || valor === "off" || valor === "0");
}

function obtenerConfigCoach() {
  const props = PropertiesService.getScriptProperties();
  return {
    model8b: props.getProperty("NUTRAFIT_COACH_MODEL_PRIMARY") || "llama-3.1-8b-instant",
    model70b: props.getProperty("NUTRAFIT_COACH_MODEL_SUMMARY") || "llama-3.3-70b-versatile",
    maxAutoPorDia: Number(props.getProperty("NUTRAFIT_COACH_MAX_AUTO_DIA") || 4),
    maxChatPorDia: Number(props.getProperty("NUTRAFIT_COACH_MAX_CHAT_DIA") || 4),
    maxTokensSalida: Number(props.getProperty("NUTRAFIT_COACH_MAX_TOKENS") || 240)
  };
}

function elegirModeloCoach(evento, payload, cfg) {
  const usar70 = Boolean(payload.usar_modelo_70b)
    || String(evento || "").toLowerCase() === "resumen_semanal";
  return usar70 ? cfg.model70b : cfg.model8b;
}

function obtenerHojaUsoCoach(ss) {
  const nombre = "coach_uso";
  let hoja = ss.getSheetByName(nombre);
  if (!hoja) {
    hoja = ss.insertSheet(nombre);
    hoja.appendRow(["timestamp", "fecha", "usuario_id", "modo", "evento", "modelo", "prompt_tokens", "completion_tokens", "total_tokens", "pregunta"]);
  }

  if (hoja.getLastRow() === 0) {
    hoja.appendRow(["timestamp", "fecha", "usuario_id", "modo", "evento", "modelo", "prompt_tokens", "completion_tokens", "total_tokens", "pregunta"]);
  }

  return hoja;
}

function validarCuotaCoachDiaria(ss, usuarioId, modo, cfg) {
  const hoja = obtenerHojaUsoCoach(ss);
  const fechaHoy = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  const max = modo === "chat" ? cfg.maxChatPorDia : cfg.maxAutoPorDia;

  if (hoja.getLastRow() <= 1) {
    return { ok: true, used: 0, max: max };
  }

  const datos = hoja.getRange(2, 1, hoja.getLastRow() - 1, 10).getValues();
  let usados = 0;
  datos.forEach(fila => {
    const fecha = String(fila[1] || "").trim();
    const uid = normalizarEmail(fila[2] || "");
    const m = String(fila[3] || "").trim();
    if (fecha === fechaHoy && uid === usuarioId && m === modo) {
      usados++;
    }
  });

  return { ok: usados < max, used: usados, max: max };
}

  const hoja = obtenerHojaUsoCoach(ss);

  hoja.appendRow([

    normalizarEmail(datos.usuarioId || ""),

    String(datos.evento || "pregunta_libre"),
    String(datos.modelo || ""),
    Number(datos.promptTokens || 0),
    Number(datos.completionTokens || 0),
    Number(datos.totalTokens || 0),
    String(datos.pregunta || "").substring(0, 220)
  ]);
}

function mapearEventoAAccionNutrafit(evento, payload) {
  const accionPayload = String(payload.accion || "").trim().toLowerCase();
  if (accionPayload) return accionPayload;

  const evt = String(evento || "").trim().toLowerCase();
  if (evt === "pregunta_libre") return "pregunta";
  if (evt === "ajuste_diario") return "alerta";
  if (evt === "abrir_app" || evt === "abrir_vista") return "consejo";
  if (evt === "registro_ejercicio") return "consejo";
  return "pregunta";
}

    String(datos.pregunta || "").substring(0, 300),

  ]);
}

function obtenerMemoriaCoachUsuario(ss, usuarioId, limite) {
  const hoja = obtenerHojaMemoriaCoach(ss);
  if (hoja.getLastRow() <= 1) return [];

  const uid = normalizarEmail(usuarioId);
  const max = Math.max(1, Number(limite || 6));
  const datos = hoja.getRange(2, 1, hoja.getLastRow() - 1, 7).getValues();
  const filtrados = [];

  for (let i = datos.length - 1; i >= 0; i--) {
    const fila = datos[i];
    if (normalizarEmail(fila[2]) !== uid) continue;
    filtrados.push({
      fecha: String(fila[1] || ""),
      accion: String(fila[3] || ""),
      intencion: String(fila[4] || ""),
      pregunta: String(fila[5] || ""),
      respuesta: String(fila[6] || "")
    });
    if (filtrados.length >= max) break;
  }

  return filtrados.reverse();
}

function construirContextoUsuarioCoach(ss, usuarioId) {
  const uid = normalizarEmail(usuarioId);
  const tz = Session.getScriptTimeZone();

  const hoy = Utilities.formatDate(new Date(), tz, "yyyy-MM-dd");
  const resumen = {
    fecha_hoy: hoy,
    agua_hoy_vasos: null,
    creditos_ultimo_total: null,
    creditos_ultimo_fecha: "",
    ejercicio_ultimo: null,
    peso_ultimo: null
  };

  const hojaAgua = ss.getSheetByName("Control_Agua");
  if (hojaAgua && hojaAgua.getLastRow() > 1) {
    const datosAgua = hojaAgua.getDataRange().getValues();
    let vasosHoy = 0;
    for (let i = datosAgua.length - 1; i >= 1; i--) {
      const fila = datosAgua[i];
      if (normalizarEmail(fila[0]) !== uid) continue;
      const fechaFila = String(fila[1] || "").trim();
      if (!(fechaFila === hoy || fechaFila.indexOf(hoy) === 0)) continue;
      const vasosRaw = String(fila[2] || "");
      const m = vasosRaw.match(/(\d+)/);
      if (m) vasosHoy = Math.max(vasosHoy, Number(m[1] || 0));
    }
    resumen.agua_hoy_vasos = vasosHoy;
  }

  const hojaCred = ss.getSheetByName("Creditos") || ss.getSheetByName("CREDITOS");
  if (hojaCred && hojaCred.getLastRow() > 1) {
    const datosCred = hojaCred.getDataRange().getValues();
    for (let i = datosCred.length - 1; i >= 1; i--) {
      const fila = datosCred[i];
      if (normalizarEmail(fila[0]) !== uid) continue;
      resumen.creditos_ultimo_total = Number(fila[6]);
      resumen.creditos_ultimo_fecha = String(fila[1] || "");
      break;
    }
  }

  const hojaEj = ss.getSheetByName("ejercicio");
  if (hojaEj && hojaEj.getLastRow() > 1) {
    const datosEj = hojaEj.getDataRange().getValues();
    for (let i = datosEj.length - 1; i >= 1; i--) {
      const fila = datosEj[i];
      if (normalizarEmail(fila[0]) !== uid) continue;
      resumen.ejercicio_ultimo = {
        fecha: fila[1] instanceof Date ? Utilities.formatDate(fila[1], tz, "yyyy-MM-dd") : String(fila[1] || ""),
        actividad: String(fila[2] || ""),
        tiempo_min: Number(fila[3] || 0),
        distancia: Number(fila[5] || 0),
        pasos: Number(fila[6] || 0)
      };
      break;
    }
  }

  const hojaPeso = ss.getSheetByName("Peso");
  if (hojaPeso && hojaPeso.getLastRow() > 1) {
    const datosPeso = hojaPeso.getDataRange().getValues();
    for (let i = datosPeso.length - 1; i >= 1; i--) {
      const fila = datosPeso[i];
      if (normalizarEmail(fila[0]) !== uid) continue;
      resumen.peso_ultimo = {
        fecha: String(fila[1] || ""),
        peso: Number(fila[2] || 0),
        diferencia: String(fila[3] || "")
      };
      break;
    }
  }

  return resumen;
}

function solicitarGroqCoach(modelo, mensajeUsuario, maxTokensSalida, modo, evento) {
  const props = PropertiesService.getScriptProperties();
  const apiKey = props.getProperty("GROQ_API_KEY");
  if (!apiKey) {
    throw new Error("Falta GROQ_API_KEY en Script Properties");
  }

  const modoTxt = String(modo || "auto").toLowerCase() === "chat" ? "chat" : "auto";
  const eventoTxt = String(evento || "").toLowerCase();
  const esChatLibre = (modoTxt === "chat" || eventoTxt === "pregunta_libre");
  const temperature = esChatLibre ? 0.9 : 0.5;
  const maxTokensFinal = Math.max(Number(maxTokensSalida || 260), modoTxt === "chat" ? 320 : 260);

  const payload = {
    model: modelo,
    temperature: temperature,
    max_tokens: maxTokensFinal,
    messages: [
      { role: "system", content: promptMaestroNutrafit },
      { role: "user", content: mensajeUsuario }
    ],
    response_format: { type: "json_object" }
  };

  const response = UrlFetchApp.fetch("https://api.groq.com/openai/v1/chat/completions", {
    method: "post",
    contentType: "application/json",
    headers: {
      Authorization: "Bearer " + apiKey
    },
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  });

  const status = response.getResponseCode();
  const body = response.getContentText();
  if (status < 200 || status >= 300) {
    throw new Error("Groq HTTP " + status + " - " + body.substring(0, 500));
  }

  return JSON.parse(body || "{}");
}

function parsearRespuestaCoachJSON(texto) {
  if (!texto) return {};

  try {
    return JSON.parse(texto);
  } catch (_) {
    const match = String(texto).match(/\{[\s\S]*\}/);
    if (!match) return {};
    try {
      return JSON.parse(match[0]);
    } catch (_) {
      const textoPlano = String(texto || "").trim();
      return textoPlano ? { __textoPlano: textoPlano } : {};
    }
  }
}

function detectarIntencionCoach(pregunta, evento) {
  const texto = (String(evento || "") + " " + String(pregunta || "")).toLowerCase();

  if (/dulce|chocolate|antojo|ansiedad/.test(texto)) return "dulce";
  if (/chia|ch[ií]a|semilla|omega\s*3|fibra|beneficio|vitamina|mineral|calcio|hierro/.test(texto)) return "nutricion_general";
  if (/desayuno|media\s*ma[ñn]ana|merienda|cena|comida/.test(texto)) return "comidas";
  if (/fuera|restaurante|bar|carta|pedido|take away|takeaway/.test(texto)) return "comer_fuera";
  if (/ejercicio|entreno|entrenamiento|pasos|caminar|correr|gym|gimnasio/.test(texto)) return "ejercicio";
  if (/agua|hidrata|hidrat/.test(texto)) return "agua";
  if (/desmotiv|ansiosa|agotada|cansada|abandono|fracaso/.test(texto)) return "motivacion";
  if (/finde|fin de semana|sábado|sabado|domingo/.test(texto)) return "finde";
  if (/cr[eé]dit|pasad|exceso|compens|ajust/.test(texto)) return "creditos";

  if (String(evento || "").toLowerCase() === "registro_ejercicio") return "ejercicio";
  if (String(evento || "").toLowerCase() === "ajuste_diario") return "creditos";
  if (String(evento || "").toLowerCase() === "abrir_app") return "inicio_dia";

  return "general";
}

function normalizarRespuestaCoach(respuesta, evento, contexto, pregunta, contenidoRaw, modo) {
  const disponibles = Number(contexto.creditos_disponibles);
  const consumidos = Number(contexto.creditos_consumidos);
  const tieneCreditos = !isNaN(disponibles) && !isNaN(consumidos);
  const exceso = tieneCreditos && disponibles < 0 ? Math.abs(disponibles) : 0;

  const estado = String(respuesta.estado_dia || "").toUpperCase();
  const estadoFinal = (estado === "VERDE" || estado === "AMARILLO" || estado === "ROJO")
    ? estado
    : (exceso > 0 ? "ROJO" : (tieneCreditos && disponibles <= 3 ? "AMARILLO" : "VERDE"));

  const preguntaTxt = String(pregunta || "").trim();
  const eventoTxt = String(evento || "").trim().toLowerCase();
  const textoPlano = String(respuesta.__textoPlano || "").trim();
  const contenidoPlano = String(contenidoRaw || "").trim();
  const intencion = detectarIntencionCoach(preguntaTxt, eventoTxt);
  const modoTxt = String(modo || "auto").toLowerCase() === "chat" ? "chat" : "auto";
  const esPreguntaDirecta = (modoTxt === "chat" || eventoTxt === "pregunta_libre");
  const plantilla = construirFallbackSegunIntencion(intencion, {
    exceso: exceso,
    disponibles: disponibles,
    consumidos: consumidos,
    pregunta: preguntaTxt
  });

  const lecturaDinamica = tieneCreditos
    ? (exceso > 0
      ? "Hay un exceso en el consumo de creditos. Te ayudo a compensarlo con decisiones concretas."
      : (disponibles <= 3
        ? "Vas justa de creditos y estas a tiempo de cerrar bien el dia."
        : "Vas con buen margen para mantener equilibrio."))
    : "Reviso tu situacion y te doy el siguiente paso mas util ahora.";

  const lecturaPregunta = preguntaTxt
    ? "Sobre tu pregunta: " + preguntaTxt
    : "";

  const lecturaBase = respuesta.lectura_rapida
    ? String(respuesta.lectura_rapida)
    : (textoPlano || contenidoPlano || (lecturaPregunta ? (lecturaPregunta + ". " + lecturaDinamica) : lecturaDinamica));

  const respuestaDirecta = String(respuesta.respuesta_directa || textoPlano || "").trim();

  if (esPreguntaDirecta) {
    return {
      respuesta_directa: respuestaDirecta || (preguntaTxt ? ("Necesito un dato mas para responder con precision a tu pregunta: " + preguntaTxt) : "Necesito tu pregunta concreta para ayudarte mejor."),
      estado_dia: "",
      lectura_rapida: "",
      accion_ahora: "",
      siguiente_paso: "",
      ajuste_creditos: "",
      micro_habito: "",
      mensaje_motivador: ""
    };
  }

  let lecturaFinal = lecturaBase;
  if (exceso > 0) {
    const objetivo = Math.max(0, consumidos - exceso);
    lecturaFinal = "Hay un exceso en el consumo de creditos. Objetivo: " + objetivo + ", consumidos: " + consumidos + ", exceso: " + exceso + ". " + lecturaBase;
  }

  let siguientePaso = String(respuesta.siguiente_paso || plantilla.siguiente_paso);
  if (exceso > 0) {
    siguientePaso = "Para el siguiente paso, usa verduras cocinadas y una proteina magra. Deja crudos solo para la cena.";
  }

  const accionAhoraDefault = plantilla.accion_ahora;
  const siguientePasoDefault = plantilla.siguiente_paso;
  const microHabitoDefault = plantilla.micro_habito;
  const motivadorDefault = plantilla.mensaje_motivador;

  return {
    respuesta_directa: respuestaDirecta,
    estado_dia: estadoFinal,
    lectura_rapida: lecturaFinal,
    accion_ahora: String(respuesta.accion_ahora || accionAhoraDefault),
    siguiente_paso: String(respuesta.siguiente_paso || siguientePaso || siguientePasoDefault),
    ajuste_creditos: String(respuesta.ajuste_creditos || (exceso > 0
      ? "Compensa con ejercicio moderado hoy o manana y ajusta una comida, sin castigos."
      : "Mantiene el reparto equilibrado de creditos para llegar con margen a la noche.")),
    micro_habito: String(respuesta.micro_habito || microHabitoDefault),
    mensaje_motivador: String(respuesta.mensaje_motivador || motivadorDefault)
  };
}

/**
 * Sanitiza valores antes de guardar

  if (valor === null || valor === undefined) return "";
  if (typeof valor === "string") return valor.trim();
  return valor;
