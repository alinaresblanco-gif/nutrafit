/* ============================================================
    CONTROL TOTAL NUTRAFIT - VERSIÓN INTEGRAL (UNIFICADA)
    VERSIÓN MEJORADA CON SOPORTE PARA DESPENSA DINÁMICO
   ============================================================ */

const SPREADSHEET_ID = "1ynyoanVl3CX-a9B-I45iy1BE6vhdJ0hrrRfipziuKz8";
const ADMIN_EMAIL = "alinares.blanco@gmail.com";
const ADMIN_KEY = "Nutrafit-Admin-2026";

/**
 * doGet: Gestiona la carga de páginas HTML y el envío de datos JSON
 */
function doGet(e) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);

    if (e.parameter.action === "admin_aprobar" && e.parameter.adminKey === ADMIN_KEY && e.parameter.email) {
      return aprobarUsuarioNutrafit(e.parameter.email);
    }

    const tabla = e.parameter.tabla || e.parameter.hoja;

    // --- 1. LÓGICA DE CARGA DE VISTAS HTML ---
    if (e.parameter.v) {
      return HtmlService.createTemplateFromFile(e.parameter.v)
          .evaluate()
          .addMetaTag('viewport', 'width=device-width, initial-scale=1')
          .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
    }

    // --- 2. LÓGICA DE LECTURA DE DATOS (JSON) ---
    if (!tabla) {
      return HtmlService.createTemplateFromFile('index')
          .evaluate()
          .addMetaTag('viewport', 'width=device-width, initial-scale=1')
          .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
    }

    const tz = Session.getScriptTimeZone();

    // CASO: ALIMENTOS
    if (tabla === "alimentos") {
      try {
        const sheetNames = ["alimentos", "Alimentos", "ALIMENTOS"];
        let sheet = null;
        for (let name of sheetNames) {
          sheet = ss.getSheetByName(name);
          if (sheet) break;
        }
        if (!sheet) {
          Logger.log("Error: No se encontró la hoja de alimentos");
          return finalizarJSON([]);
        }
        const data = sheet.getDataRange().getValues();
        if (data.length <= 1) return finalizarJSON([]);
        data.shift();
        const dataLimpia = data.filter(fila => fila[0] && fila[0].toString().trim() !== "");
        Logger.log("Alimentos cargados: " + dataLimpia.length);
        return finalizarJSON(dataLimpia);
      } catch (err) {
        Logger.log("Error en alimentos: " + err.toString());
        return finalizarJSON([]);
      }
    }

    // CASO: MENÚS SEMANALES
    else if (tabla === "menus_semanales" || tabla === "menu_semanal") {
      const usuarioId = normalizarEmail(e.parameter.usuario_id);
      if (!usuarioId) return finalizarJSON([]);

      const infoUsr = obtenerUsuarioNutrafit(ss, usuarioId);
      if (!infoUsr || !infoUsr.fila || estadoUsuarioNutrafit(infoUsr.fila) !== "activo") {
        return finalizarJSON([]);
      }

      const sheet = ss.getSheetByName("menus_semanales");
      if (!sheet) return finalizarJSON([]);
      const vals = sheet.getDataRange().getValues();
      if (vals.length <= 1) return finalizarJSON([]);

      const cabeceras = vals.shift();

      const resultado = vals
        .filter(fila => normalizarEmail(fila[0]) === usuarioId)
        .map(fila => {
          const obj = {};
          cabeceras.forEach((cab, i) => {
            const celda = fila[i];
            if (celda instanceof Date) {
              obj[cab] = Utilities.formatDate(celda, tz, "yyyy-MM-dd");
            } else {
              obj[cab] = (celda !== null && celda !== undefined) ? celda : "";
            }
          });
          return obj;
        });

      return finalizarJSON(resultado);
    }

    // CASO: RECETAS
    else if (tabla === "recetas") {
      const sheetRec = ss.getSheetByName("recetas");
      if (!sheetRec) return finalizarJSON([]);
      const datos = sheetRec.getDataRange().getValues();
      if (datos.length <= 1) return finalizarJSON([]);
      datos.shift();
      const filasLimpias = datos.map(fila => fila.map(celda => {
        if (celda === null || celda === undefined) return "";
        if (celda instanceof Date) return celda.toLocaleDateString();
        return celda.toString();
      }));
      return finalizarJSON(filasLimpias);
    }

    // CASO: EJERCICIO
    else if (tabla === "ejercicio") {
      const usuarioId = normalizarEmail(e.parameter.usuario_id);
      if (!usuarioId) return finalizarJSON([]);

      const infoUsr = obtenerUsuarioNutrafit(ss, usuarioId);
      if (!infoUsr || !infoUsr.fila || estadoUsuarioNutrafit(infoUsr.fila) !== "activo") {
        return finalizarJSON([]);
      }

      const sheet = ss.getSheetByName("ejercicio");
      if (!sheet) return finalizarJSON([]);
      const data = sheet.getDataRange().getValues();
      if (data.length <= 1) return finalizarJSON([]);
      data.shift();

      // Columnas: [usuario_id, fecha, actividad, tiempo, foto, distancia, pasos, desnivel, vel_media]
      // Añadimos rowId al final para permitir borrado exacto desde frontend.
      const dataUsuario = data
        .map((fila, idx) => ({ fila, rowId: idx + 2 }))
        .filter(item => normalizarEmail(item.fila[0]) === usuarioId)
        .map(item => {
          const filaConvertida = item.fila.map(celda =>
            celda instanceof Date
              ? Utilities.formatDate(celda, tz, "yyyy-MM-dd'T'HH:mm:ss")
              : celda
          );
          filaConvertida.push(item.rowId);
          return filaConvertida;
        });
      return finalizarJSON(dataUsuario.slice(-20).reverse());
    }

    // CASO: PESO
    else if (tabla === "peso") {
      const usuarioId = normalizarEmail(e.parameter.usuario_id);
      if (!usuarioId) return finalizarJSON([]);

      const infoUsr = obtenerUsuarioNutrafit(ss, usuarioId);
      if (!infoUsr || !infoUsr.fila || estadoUsuarioNutrafit(infoUsr.fila) !== "activo") {
        return finalizarJSON([]);
      }

      const sheet = ss.getSheetByName("Peso");
      if (!sheet) return finalizarJSON([]);
      const data = sheet.getDataRange().getValues();
      if (data.length <= 1) return finalizarJSON([]);
      data.shift();

      // Añadimos rowId al final para permitir borrado exacto desde frontend.
      const dataUsuario = data
        .map((fila, idx) => ({ fila, rowId: idx + 2 }))
        .filter(item => normalizarEmail(item.fila[0]) === usuarioId)
        .map(item => {
          const filaConvertida = item.fila.map(celda =>
            celda instanceof Date ? Utilities.formatDate(celda, tz, "yyyy-MM-dd") : celda
          );
          filaConvertida.push(item.rowId);
          return filaConvertida;
        });
      return finalizarJSON(dataUsuario.slice(-15).reverse());
    }

    // CASO: CRÉDITOS
    else if (tabla === "creditos" || tabla === "CREDITOS") {
      const usuarioId = normalizarEmail(e.parameter.usuario_id);
      if (!usuarioId) return finalizarJSON([]);

      // Verificar usuario activo (igual que agua)
      const infoUsr = obtenerUsuarioNutrafit(ss, usuarioId);
      if (!infoUsr || !infoUsr.fila || estadoUsuarioNutrafit(infoUsr.fila) !== "activo") {
        return finalizarJSON([]);
      }

      const sheet = ss.getSheetByName("Creditos") || ss.getSheetByName("CREDITOS");
      if (!sheet) return finalizarJSON([]);
      const data = sheet.getDataRange().getValues();
      if (data.length <= 1) return finalizarJSON([]);
      data.shift();

      // Filtrar por usuario y convertir fechas
      const dataUsuario = data
        .filter(fila => normalizarEmail(fila[0]) === usuarioId)
        .map(fila => fila.map(celda =>
          celda instanceof Date ? Utilities.formatDate(celda, tz, "yyyy-MM-dd") : celda
        ));
      return finalizarJSON(dataUsuario.slice(-10).reverse());
    }

    // CASO: AGUA
    else if (tabla === "agua") {
      const usuarioId = normalizarEmail(e.parameter.usuario_id);

      // Para lectura solo se exige que el usuario exista y esté activo.
      // El control de dispositivo se aplica únicamente en escritura (doPost).
      if (!usuarioId) return finalizarJSON([]);
      const infoUsuario = obtenerUsuarioNutrafit(ss, usuarioId);
      if (!infoUsuario || !infoUsuario.fila || estadoUsuarioNutrafit(infoUsuario.fila) !== "activo") {
        return finalizarJSON([]);
      }

      const sheet = ss.getSheetByName("Control_Agua");
      if (!sheet) return finalizarJSON([]);
      const data = sheet.getDataRange().getValues();
      if (data.length <= 1) return finalizarJSON([]);
      data.shift();
      const dataUsuario = data.filter(fila => {
        // Nuevo formato: [usuario_id, fecha, vasos, estado, hora]
        if (Array.isArray(fila) && fila.length >= 5) {
          return normalizarEmail(fila[0]) === usuarioId;
        }

        // Formato legado: [fecha, vasos, estado, hora]
        // No tenía usuario_id, así que se mantiene compatibilidad para no perder historial previo.
        return Array.isArray(fila) && fila.length >= 4;
      });
      return finalizarJSON(dataUsuario.slice(-10).reverse());
    }

    else {
      return finalizarJSON([]);
    }

  } catch (err) {
    Logger.log("Error en doGet: " + err.toString());
    return ContentService.createTextOutput(JSON.stringify({ error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * doPost: Gestión de guardado de datos
 */
function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var tipo = payload.tipo || payload.accion || payload.action;

    if (tipo === "verificar_acceso") {
      return verificarAccesoNutrafit(ss, payload);
    }

    if (tipo === "solicitar_acceso") {
      return solicitarAccesoNutrafit(ss, payload);
    }

    if (tipo === "coach_nutrafit") {
      return procesarCoachNutrafit(ss, payload);
    }

    if (tipo === "ejecutar_mini_test_coach") {
      return ejecutarMiniTestCoach(ss, payload);
    }


    // REGISTRO DE USUARIO
    if (tipo === "registro_usuario") {
      const hojaUsuarios = ss.getSheetByName("usuarios");
      const { usuario_id, nombre, dispositivo_id } = payload;
      const datos = hojaUsuarios.getDataRange().getValues();
      const existe = datos.findIndex(fila => fila[0] === usuario_id);
      if (existe !== -1) {
        return ContentService.createTextOutput(JSON.stringify({ status: "existe" })).setMimeType(ContentService.MimeType.JSON);
      }
      const fecha_registro = new Date();
      hojaUsuarios.appendRow([
        usuario_id,
        nombre,
        Utilities.formatDate(fecha_registro, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"),
        false,
        dispositivo_id,
        "",
        false
      ]);
      // Notificar al admin
      MailApp.sendEmail({
        to: "alinares.blanco@gmail.com", // <-- CAMBIA POR TU CORREO
        subject: "Nuevo registro en Nutrafit",
        body: `Nuevo usuario registrado: ${nombre} (${usuario_id})\n\nValídalo en la hoja de usuarios.`
      });
      return ContentService.createTextOutput(JSON.stringify({ status: "pendiente" })).setMimeType(ContentService.MimeType.JSON);
    }

    // LOGIN DE USUARIO
    if (tipo === "login_usuario") {
      const hojaUsuarios = ss.getSheetByName("usuarios");
      const { usuario_id, dispositivo_id } = payload;
      const datos = hojaUsuarios.getDataRange().getValues();
      const fila = datos.find(fila => fila[0] === usuario_id);
      if (!fila) {
        return ContentService.createTextOutput(JSON.stringify({ status: "no_registrado" })).setMimeType(ContentService.MimeType.JSON);
      }
      if (!fila[3]) {
        return ContentService.createTextOutput(JSON.stringify({ status: "no_validado" })).setMimeType(ContentService.MimeType.JSON);
      }
      if (fila[4] !== dispositivo_id) {
        return ContentService.createTextOutput(JSON.stringify({ status: "otro_dispositivo" })).setMimeType(ContentService.MimeType.JSON);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "ok", nombre: fila[1] })).setMimeType(ContentService.MimeType.JSON);
    }

    // NOTIFICAR VALIDACIÓN DE USUARIO (ADMIN)
    if (tipo === "notificar_validacion") {
      const hojaUsuarios = ss.getSheetByName("usuarios");
      const { usuario_id } = payload;
      const datos = hojaUsuarios.getDataRange().getValues();
      const idx = datos.findIndex(fila => fila[0] === usuario_id);
      if (idx === -1) {
        return ContentService.createTextOutput("Usuario no encontrado");
      }
      const fila = datos[idx];
      if (fila[3] && !fila[5]) {
        // Solo si está validado y no se ha notificado
        const fecha_validacion = new Date();
        hojaUsuarios.getRange(idx + 1, 6).setValue(Utilities.formatDate(fecha_validacion, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"));
        // Notificar al usuario
        MailApp.sendEmail({
          to: fila[0],
          subject: "Nutrafit: Usuario validado",
          body: `Hola ${fila[1]},\n\nTu usuario ha sido validado. Ya puedes acceder a Nutrafit.\n\n¡Bienvenida/o!`
        });
        return ContentService.createTextOutput("Notificado");
      }
      return ContentService.createTextOutput("No validado o ya notificado");
    }

    // GUARDAR EJERCICIO
    if (tipo === "guardar_ejercicio") {
      const sheetEj = ss.getSheetByName("ejercicio");
      const usuarioId = normalizarEmail(payload.usuario_id);
      const dispositivoId = String(payload.dispositivo_id || "").trim();

      if (!sheetEj) {
        return ContentService.createTextOutput("No existe la hoja ejercicio");
      }

      if (!usuarioId || !dispositivoId || !sesionValidaNutrafit(ss, usuarioId, dispositivoId)) {
        return ContentService.createTextOutput("Sesión inválida");
      }

      var celdaFoto = (payload.imagenBase64 && payload.imagenBase64.length > 100) ? "data:image/jpeg;base64," + payload.imagenBase64 : "";
      var distLim = parseFloat(String(payload.distancia || "0").replace("'", "").replace(",", "."));
      var tiempoLim = parseFloat(String(payload.tiempo || "0").replace("'", "").replace(",", "."));
      var velMedia = (distLim > 0 && tiempoLim > 0) ? (distLim / (tiempoLim / 60)).toFixed(2) : 0;

      sheetEj.appendRow([usuarioId, new Date(), payload.actividad, tiempoLim, celdaFoto, distLim, payload.pasos || 0, payload.desnivel || 0, velMedia]);
      return ContentService.createTextOutput("Éxito Ejercicio");
    }

    // GUARDAR RECETA
    else if (tipo === "guardar_receta") {
      var sheetRec = ss.getSheetByName("recetas") || ss.insertSheet("recetas");
      var celdaFotoRec = (payload.imagenBase64 && payload.imagenBase64.length > 100) ? "data:image/jpeg;base64," + payload.imagenBase64 : "";
      sheetRec.appendRow([payload.usuario_id || "comun", new Date(), celdaFotoRec, payload.nombre, payload.categoria, payload.ingredientes, payload.elaboracion]);
      return ContentService.createTextOutput("Éxito Receta");
    }

    // GUARDAR CRÉDITOS
    else if (tipo === "creditos") {
      var sheetCreditos = ss.getSheetByName("Creditos") || ss.getSheetByName("CREDITOS");
      sheetCreditos.appendRow([payload.usuario_id, payload.fecha, payload.genero, payload.edad, payload.peso, payload.altura, payload.total]);
      return ContentService.createTextOutput("Éxito Créditos");
    }

    // GUARDAR MENÚ SEMANAL
    else if (tipo === "guardar_menu_semanal" || tipo === "guardarFila" || tipo === "actualizar_menu_semanal") {
      var sheetMenus = ss.getSheetByName("menus_semanales") || ss.insertSheet("menus_semanales");
      const usuarioId = normalizarEmail(payload.usuario_id);
      const dispositivoId = String(payload.dispositivo_id || "").trim();
      const esActualizacion = (tipo === "actualizar_menu_semanal") || (String(payload.modo || "").toLowerCase() === "actualizar");

      if (!usuarioId || !dispositivoId || !sesionValidaNutrafit(ss, usuarioId, dispositivoId)) {
        return ContentService.createTextOutput("Sesión inválida");
      }

      // ✅ Añadimos usuario_id como primera columna
      const headers = ["usuario_id", "Semana inicio", "Fecha", "Día", "Momento", "Ingrediente", "Cantidad", "Puntos", "Notas", "Presupuesto"];

      if (sheetMenus.getLastRow() === 0) {
        sheetMenus.appendRow(headers);
      } else {
        const filaCabecera = sheetMenus.getRange(1, 1, 1, headers.length).getValues()[0];
        if (filaCabecera.join("|") !== headers.join("|")) {
          sheetMenus.getRange(1, 1, 1, headers.length).setValues([headers]);
        }
      }

      const filas = Array.isArray(payload.filas)
        ? payload.filas
        : Array.isArray(payload.datos)
          ? payload.datos
          : [payload.datos || payload];

      const filasParaGuardar = filas
        .filter(f => f && typeof f === 'object' && Object.keys(f).length > 0)
        .map(f => [
          usuarioId,
          sanitizeCampo(f["Semana inicio"] || f.semana_inicio || f.semana || ""),
          sanitizeCampo(f["Fecha"]         || f.fecha         || ""),
          sanitizeCampo(f["Día"]           || f.Dia           || f.dia || ""),
          sanitizeCampo(f["Momento"]       || f.momento       || ""),
          sanitizeCampo(f["Ingrediente"]   || f.ingrediente   || ""),
          "",
          sanitizeCampo(f["Puntos"]        || f.puntos        || ""),
          sanitizeCampo(f["Notas"]         || f.notas         || ""),
          sanitizeCampo(f["Presupuesto"]   || f.presupuesto   || payload.presupuesto || "")
        ]);

      // Evita duplicados en el propio payload antes de tocar la hoja.
      const filasUnicasMap = {};
      filasParaGuardar.forEach(fila => {
        const clave = [fila[1], fila[2], fila[3], fila[4], fila[5]]
          .map(valor => String(valor || "").trim().toLowerCase())
          .join("|");
        filasUnicasMap[clave] = fila;
      });
      const filasParaGuardarUnicas = Object.keys(filasUnicasMap).map(clave => filasUnicasMap[clave]);

      function claveFilaMenu(fila) {
        // Clave lógica de una comida dentro de la semana (ignora mayúsculas/minúsculas).
        return [fila[1], fila[2], fila[3], fila[4], fila[5]]
          .map(valor => String(valor || "").trim().toLowerCase())
          .join("|");
      }

      // Semanas incluidas en esta operación.
      const semanasAReemplazar = Array.from(new Set(
        filasParaGuardarUnicas
          .map(fila => String(fila[1] || '').trim()) // Columna "Semana inicio"
          .filter(valor => valor !== '')
      ));

      if (semanasAReemplazar.length > 0 && sheetMenus.getLastRow() > 1) {
        const dataActual = sheetMenus.getDataRange().getValues();

        if (esActualizacion) {
          // Actualizar semana activa: reemplazo total por usuario+semana para evitar duplicados acumulados.
          const filasAEliminar = [];
          dataActual.slice(1).forEach((fila, idx) => {
            const filaUsuario = normalizarEmail(fila[0]);
            const filaSemanaInicio = String(fila[1] || '').trim();
            if (filaUsuario === usuarioId && semanasAReemplazar.includes(filaSemanaInicio)) {
              filasAEliminar.push(idx + 2); // row number real en la hoja
            }
          });

          filasAEliminar.sort((a, b) => b - a).forEach(rowNum => sheetMenus.deleteRow(rowNum));

          if (filasParaGuardarUnicas.length > 0) {
            sheetMenus.getRange(
              sheetMenus.getLastRow() + 1, 1,
              filasParaGuardarUnicas.length, headers.length
            ).setValues(filasParaGuardarUnicas);
          }
        } else {
          // Guardar semana nueva: no sobrescribe una semana existente.
          const existeSemana = dataActual.slice(1).some(fila => {
            const filaUsuario = normalizarEmail(fila[0]);
            const filaSemanaInicio = String(fila[1] || '').trim();
            return filaUsuario === usuarioId && semanasAReemplazar.includes(filaSemanaInicio);
          });

          if (existeSemana) {
            return ContentService
              .createTextOutput("Ya existe una semana con esa fecha. Usa Actualizar para modificarla.")
              .setMimeType(ContentService.MimeType.TEXT);
          }

          if (filasParaGuardarUnicas.length > 0) {
            sheetMenus.getRange(
              sheetMenus.getLastRow() + 1, 1,
              filasParaGuardarUnicas.length, headers.length
            ).setValues(filasParaGuardarUnicas);
          }
        }
      } else if (filasParaGuardarUnicas.length > 0) {
        // Primera carga (sin semanas previas) o sin colisiones.
        sheetMenus.getRange(
          sheetMenus.getLastRow() + 1, 1,
          filasParaGuardarUnicas.length, headers.length
        ).setValues(filasParaGuardarUnicas);
      }

      return ContentService.createTextOutput(esActualizacion ? "Éxito Actualizar Menú Semanal" : "Éxito Menú Semanal");
    }

    // GUARDAR PESO
    else if (tipo === "peso") {
      const sheetPeso = ss.getSheetByName("Peso");
      sheetPeso.appendRow([payload.usuario_id, payload.fecha, payload.peso, payload.diferencia]);
      return ContentService.createTextOutput("Éxito Peso");
    }

    // ELIMINAR REGISTRO DE PESO
    else if (tipo === "eliminar_peso") {
      const sheetPeso = ss.getSheetByName("Peso");
      const usuarioId = normalizarEmail(payload.usuario_id);
      const dispositivoId = String(payload.dispositivo_id || "").trim();
      const rowId = Number(payload.row_id || 0);
      const fecha = String(payload.fecha || "").trim();
      const peso = String(payload.peso || "").trim();

      if (!sheetPeso) {
        return ContentService.createTextOutput("No existe la hoja Peso");
      }

      if (!usuarioId || !dispositivoId || !sesionValidaNutrafit(ss, usuarioId, dispositivoId)) {
        return ContentService.createTextOutput("Sesión inválida");
      }

      // 1) Intento principal: borrar por row_id exacto (viene de doGet peso)
      if (rowId > 1 && rowId <= sheetPeso.getLastRow()) {
        const fila = sheetPeso.getRange(rowId, 1, 1, 4).getValues()[0];
        if (normalizarEmail(fila[0]) === usuarioId) {
          sheetPeso.deleteRow(rowId);
          return ContentService.createTextOutput("Éxito Eliminar Peso");
        }
      }

      // 2) Fallback: buscar por usuario + fecha + peso desde abajo (más reciente primero)
      const data = sheetPeso.getDataRange().getValues();
      for (let i = data.length - 1; i >= 1; i--) {
        const fila = data[i];
        const filaUsuario = normalizarEmail(fila[0]);
        const filaFecha = String(fila[1] || "").trim();
        const filaPeso = String(fila[2] || "").trim();

        if (filaUsuario === usuarioId && filaFecha === fecha && filaPeso === peso) {
          sheetPeso.deleteRow(i + 1);
          return ContentService.createTextOutput("Éxito Eliminar Peso");
        }
      }

      return ContentService.createTextOutput("Registro no encontrado");
    }

    // ELIMINAR REGISTRO DE EJERCICIO
    else if (tipo === "eliminar_ejercicio") {
      const sheetEj = ss.getSheetByName("ejercicio");
      const usuarioId = normalizarEmail(payload.usuario_id);
      const dispositivoId = String(payload.dispositivo_id || "").trim();
      const rowId = Number(payload.row_id || 0);
      const fechaVal = String(payload.fecha || "").trim();
      const actividad = String(payload.actividad || "").trim();

      if (!sheetEj) {
        return ContentService.createTextOutput("No existe la hoja ejercicio");
      }

      if (!usuarioId || !dispositivoId || !sesionValidaNutrafit(ss, usuarioId, dispositivoId)) {
        return ContentService.createTextOutput("Sesión inválida");
      }

      // 1) Intento principal: borrar por row_id exacto
      if (rowId > 1 && rowId <= sheetEj.getLastRow()) {
        const fila = sheetEj.getRange(rowId, 1, 1, 3).getValues()[0];
        if (normalizarEmail(fila[0]) === usuarioId) {
          sheetEj.deleteRow(rowId);
          return ContentService.createTextOutput("Éxito Eliminar Ejercicio");
        }
      }

      // 2) Fallback: buscar por usuario + fecha + actividad desde abajo
      const data = sheetEj.getDataRange().getValues();
      for (let i = data.length - 1; i >= 1; i--) {
        const fila = data[i];
        const filaUsuario = normalizarEmail(fila[0]);
        const filaFecha = String(fila[1] instanceof Date
          ? Utilities.formatDate(fila[1], Session.getScriptTimeZone(), "yyyy-MM-dd'T'HH:mm:ss")
          : fila[1] || "").trim();
        const filaActividad = String(fila[2] || "").trim();
        if (filaUsuario === usuarioId && filaFecha.startsWith(fechaVal.substring(0, 10)) && filaActividad === actividad) {
          sheetEj.deleteRow(i + 1);
          return ContentService.createTextOutput("Éxito Eliminar Ejercicio");
        }
      }

      return ContentService.createTextOutput("Registro no encontrado");
    }

    // GUARDAR ALIMENTO NUEVO
    else if (tipo === "guardar_alimento") {
      const sheetAlim = ss.getSheetByName("Alimentos") || ss.getSheetByName("alimentos") || ss.insertSheet("Alimentos");
      sheetAlim.appendRow([payload.nombre, payload.grupo, payload.proteinas, payload.carbohidratos, payload.grasas, payload.fibra, payload.manual, payload.calculado, payload.neto, new Date()]);
      return ContentService.createTextOutput("Éxito Alimento");
    }

    // GUARDAR AGUA (POR DEFECTO)
    else {
      const sheetAgua = ss.getSheetByName("Control_Agua");
      const usuarioId = normalizarEmail(payload.usuario_id);
      const dispositivoId = String(payload.dispositivo_id || "").trim();

      if (!sheetAgua) {
        return ContentService.createTextOutput("No existe la hoja Control_Agua");
      }

      if (!usuarioId || !dispositivoId || !sesionValidaNutrafit(ss, usuarioId, dispositivoId)) {
        return ContentService.createTextOutput("Sesión inválida");
      }

      const fechaActual = new Date();
      sheetAgua.appendRow([
        usuarioId,
        fechaActual.toLocaleDateString(),
        (payload.vasos || 0) + " vasos",
        (payload.vasos >= 8) ? "COMPLETADO" : "PENDIENTE",
        fechaActual.toLocaleTimeString()
      ]);
      return ContentService.createTextOutput("Éxito Agua");
    }

  } catch (err) {
    Logger.log("Error en doPost: " + err.toString());
    return ContentService.createTextOutput(JSON.stringify({ error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * Función específica para google.script.run
 */
function obtenerAlimentosDespensa() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const hoja = ss.getSheetByName("alimentos") || ss.getSheetByName("Alimentos");
    if (!hoja) return [];
    const datos = hoja.getDataRange().getValues();
    const cabeceras = datos.shift();
    const idxNombre = cabeceras.indexOf("Nombre");
    const idxNetos = cabeceras.indexOf("Netos");
    if (idxNombre === -1 || idxNetos === -1) return [];
    return datos
      .map(fila => ({ nombre: fila[idxNombre], netos: fila[idxNetos] }))
      .filter(item => item.nombre !== "" && item.nombre !== undefined);
  } catch (e) {
    Logger.log("Error obtenerAlimentosDespensa: " + e.toString());
    return [];
  }
}

function asegurarCabeceraUsuarios(hojaUsuarios) {
  const cabeceraEsperada = ["usuario_id", "nombre", "fecha_registro", "validado", "dispositivo_id", "fecha_validacion", "admin"];
  const ultimaFila = hojaUsuarios.getLastRow();

  if (ultimaFila === 0) {
    hojaUsuarios.appendRow(cabeceraEsperada);
    return;
  }

  const cabeceraActual = hojaUsuarios.getRange(1, 1, 1, cabeceraEsperada.length).getValues()[0];
  const normalizada = cabeceraActual.map(valor => String(valor || "").trim().toLowerCase());
  const esperada = cabeceraEsperada.map(valor => String(valor || "").trim().toLowerCase());

  if (normalizada.join("|") !== esperada.join("|")) {
    hojaUsuarios.getRange(1, 1, 1, cabeceraEsperada.length).setValues([cabeceraEsperada]);
  }
}

function normalizarEmail(valor) {
  return String(valor || "").trim().toLowerCase();
}

function obtenerUsuarioNutrafit(ss, email) {
  const hojaUsuarios = ss.getSheetByName("usuarios");
  if (!hojaUsuarios) return null;

  asegurarCabeceraUsuarios(hojaUsuarios);

  const datos = hojaUsuarios.getDataRange().getValues();
  const emailBuscado = normalizarEmail(email);

  for (let i = 1; i < datos.length; i++) {
    const fila = datos[i];
    if (normalizarEmail(fila[0]) === emailBuscado) {
      return { hojaUsuarios, datos, fila, indice: i };
    }
  }

  return { hojaUsuarios, datos, fila: null, indice: -1 };
}

function sesionValidaNutrafit(ss, usuarioId, dispositivoId) {
  const usuario = normalizarEmail(usuarioId);
  const dispositivo = String(dispositivoId || "").trim();
  if (!usuario || !dispositivo) return false;

  const info = obtenerUsuarioNutrafit(ss, usuario);
  if (!info || !info.fila) return false;

  const estado = estadoUsuarioNutrafit(info.fila);
  if (estado !== "activo") return false;

  const dispositivoGuardado = String(info.fila[4] || "").trim();

  // Primera asociación de dispositivo para usuarios activos sin dispositivo registrado.
  if (!dispositivoGuardado) {
    info.hojaUsuarios.getRange(info.indice + 1, 5).setValue(dispositivo);
    return true;
  }

  return dispositivoGuardado === dispositivo;
}

function estadoUsuarioNutrafit(fila) {
  if (!fila) return "no_existe";
  const estadoColumna = fila[3];
  if (estadoColumna === true || estadoColumna === "true") return "activo";
  if (String(estadoColumna).toLowerCase() === "bloqueado") return "bloqueado";
  if (String(estadoColumna).toLowerCase() === "activo") return "activo";
  if (String(fila[0] || "").trim() === "") return "no_existe";
  return "pendiente";
}

function verificarAccesoNutrafit(ss, payload) {
  const email = normalizarEmail(payload.email || payload.usuario_id);
  const dispositivoId = String(payload.dispositivo_id || "").trim();
  if (!email) {
    return finalizarJSON({ status: "error", mensaje: "Email requerido" });
  }

  const info = obtenerUsuarioNutrafit(ss, email);
  if (!info || !info.fila) {
    return finalizarJSON({ status: "no_existe" });
  }

  const estado = estadoUsuarioNutrafit(info.fila);
  if (estado === "activo") {
    // Sincroniza el dispositivo activo para que el guardado posterior de datos no falle por sesión.
    if (dispositivoId) {
      const dispositivoGuardado = String(info.fila[4] || "").trim();
      if (dispositivoGuardado !== dispositivoId) {
        info.hojaUsuarios.getRange(info.indice + 1, 5).setValue(dispositivoId);
      }
    }

    info.hojaUsuarios.getRange(info.indice + 1, 6).setValue(Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"));
    return finalizarJSON({
      status: "activo",
      usuario: {
        email: info.fila[0],
        nombre: info.fila[1] || ""
      }
    });
  }

  if (estado === "bloqueado") {
    return finalizarJSON({ status: "bloqueado" });
  }

  return finalizarJSON({ status: "pendiente" });
}

function solicitarAccesoNutrafit(ss, payload) {
  const email = normalizarEmail(payload.email || payload.usuario_id);
  const nombre = String(payload.nombre || "").trim();

  if (!email || !nombre) {
    return finalizarJSON({ status: "error", mensaje: "Email y nombre son obligatorios" });
  }

  const info = obtenerUsuarioNutrafit(ss, email);
  if (!info || !info.hojaUsuarios) {
    return finalizarJSON({ status: "error", mensaje: "No existe la hoja de usuarios" });
  }

  const estado = info.fila ? estadoUsuarioNutrafit(info.fila) : "no_existe";
  if (estado === "activo") {
    return finalizarJSON({ status: "activo" });
  }

  if (estado === "pendiente") {
    enviarCorreoSolicitudAdminNutrafit(email, nombre, new Date().toISOString(), ss, info.hojaUsuarios, info.indice);
    return finalizarJSON({ status: "pendiente", reenviado: true });
  }

  if (info.fila) {
    return finalizarJSON({ status: estado });
  }

  const fecha_registro = new Date();
  info.hojaUsuarios.appendRow([
    email,
    nombre,
    Utilities.formatDate(fecha_registro, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"),
    false,
    "",
    "",
    false
  ]);

  enviarCorreoSolicitudAdminNutrafit(email, nombre, Utilities.formatDate(fecha_registro, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"), ss);

  return finalizarJSON({ status: "pendiente" });
}

function enviarCorreoSolicitudAdminNutrafit(email, nombre, fechaIso) {
  const urlBase = ScriptApp.getService().getUrl();
  const urlAprobar = urlBase
    ? urlBase + "?action=admin_aprobar&email=" + encodeURIComponent(email) + "&adminKey=" + encodeURIComponent(ADMIN_KEY)
    : "";

  MailApp.sendEmail({
    to: ADMIN_EMAIL,
    subject: "Nuevo registro en Nutrafit: " + nombre,
    body:
      "Nueva solicitud de acceso\n" +
      "Nombre: " + nombre + "\n" +
      "Email: " + email + "\n" +
      "Fecha: " + fechaIso + "\n\n" +
      (urlAprobar ? ("Aprobar: " + urlAprobar + "\n") : "")
  });
}

function aprobarUsuarioNutrafit(email) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const info = obtenerUsuarioNutrafit(ss, email);
  if (!info || !info.fila) {
    return textoPlano("Usuario no encontrado");
  }

  info.hojaUsuarios.getRange(info.indice + 1, 4).setValue(true);
  info.hojaUsuarios.getRange(info.indice + 1, 6).setValue(Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"));
  info.hojaUsuarios.getRange(info.indice + 1, 7).setValue(ADMIN_EMAIL);

  MailApp.sendEmail({
    to: normalizarEmail(email),
    subject: "Nutrafit: acceso aprobado",
    body: "Hola " + (info.fila[1] || "") + ",\n\nTu acceso ha sido aprobado. Ya puedes entrar en Nutrafit."
  });

  return textoPlano("Usuario aprobado y notificado");
}

const HOJA_BASE_ALIMENTOS_COACH = "base_alimentos_coach";
const CABECERA_BASE_ALIMENTOS_COACH = [
  "nombre_canonico",
  "aliases",
  "categoria",

  "beneficios",
  "precauciones",
  "porcion_orientativa",

function serializarListaCoach_(valor) {
  if (!Array.isArray(valor) || !valor.length) return "";
  return valor
    .map(function(item) { return String(item || "").trim(); })
    .filter(function(item) { return item.length > 0; })
    .join(" | ");
}

function parsearListaCoach_(valor) {
  var txt = String(valor || "").trim();
  if (!txt) return [];

  var partes = txt.indexOf("|") >= 0 ? txt.split("|") : txt.split(/[;,]/);
  return partes
    .map(function(item) { return String(item || "").trim(); })
    .filter(function(item) { return item.length > 0; });
}

function asegurarHojaBaseAlimentosCoach_(ss) {
  var hoja = ss.getSheetByName(HOJA_BASE_ALIMENTOS_COACH);
  if (!hoja) {
    hoja = ss.insertSheet(HOJA_BASE_ALIMENTOS_COACH);
  }

  if (hoja.getLastRow() === 0) {
    hoja.appendRow(CABECERA_BASE_ALIMENTOS_COACH);
  } else {
    var cabActual = hoja.getRange(1, 1, 1, CABECERA_BASE_ALIMENTOS_COACH.length).getValues()[0];
    var cabActualNorm = cabActual.map(function(v) { return normalizarClaveAlimentoCoach(v); });
    var cabEsperadaNorm = CABECERA_BASE_ALIMENTOS_COACH.map(function(v) { return normalizarClaveAlimentoCoach(v); });
    if (cabActualNorm.join("|") !== cabEsperadaNorm.join("|")) {
      hoja.getRange(1, 1, 1, CABECERA_BASE_ALIMENTOS_COACH.length).setValues([CABECERA_BASE_ALIMENTOS_COACH]);
    }
  }

  if (hoja.getLastRow() <= 1) {
    sembrarBaseAlimentosCoachDesdeCatalogo_(hoja);
  }

  return hoja;
}

function sembrarBaseAlimentosCoachDesdeCatalogo_(hoja) {
  if (!hoja || !Array.isArray(CATALOGO_ALIMENTOS_COACH) || !CATALOGO_ALIMENTOS_COACH.length) return;

  var filas = CATALOGO_ALIMENTOS_COACH.map(function(item, idx) {
    return [
      String(item.nombre || "").trim(),
      serializarListaCoach_(item.aliases),
      String(item.categoria || "").trim(),
      serializarListaCoach_(item.propiedades),
      serializarListaCoach_(item.beneficios),
      serializarListaCoach_(item.precauciones),
      String(item.porcion_referencia || "").trim(),
      serializarListaCoach_(item.fuentes),
      idx + 1
    ];
  }).filter(function(fila) {
    return String(fila[0] || "").trim().length > 0;
  });

  if (!filas.length) return;

  hoja.getRange(2, 1, filas.length, CABECERA_BASE_ALIMENTOS_COACH.length).setValues(filas);
}

function mapearFilaBaseAlimentosCoach_(fila) {
  var nombre = String(fila[0] || "").trim();
  if (!nombre) return null;

  var aliases = parsearListaCoach_(fila[1]);
  if (!aliases.length) aliases = [nombre];

  var prioridad = Number(fila[8] || 9999);
  if (isNaN(prioridad)) prioridad = 9999;

  return {
    nombre: nombre,
    aliases: aliases,
    categoria: String(fila[2] || "").trim(),
    propiedades: parsearListaCoach_(fila[3]),
    beneficios: parsearListaCoach_(fila[4]),
    precauciones: parsearListaCoach_(fila[5]),
    porcion_referencia: String(fila[6] || "").trim(),
    fuentes: parsearListaCoach_(fila[7]),
    prioridad: prioridad
  };
}

function obtenerCatalogoAlimentosCoach(ss) {
  try {
    var hoja = asegurarHojaBaseAlimentosCoach_(ss);
    if (!hoja || hoja.getLastRow() <= 1) return CATALOGO_ALIMENTOS_COACH;

    var data = hoja.getRange(2, 1, hoja.getLastRow() - 1, CABECERA_BASE_ALIMENTOS_COACH.length).getValues();
    var items = data
      .map(mapearFilaBaseAlimentosCoach_)
      .filter(function(item) { return item !== null; })
      .sort(function(a, b) {
        return Number(a.prioridad || 9999) - Number(b.prioridad || 9999);
      });

    return items.length ? items : CATALOGO_ALIMENTOS_COACH;
  } catch (e) {
    Logger.log("Error obtenerCatalogoAlimentosCoach: " + e.toString());
    return CATALOGO_ALIMENTOS_COACH;
  }
}

function procesarCoachNutrafit(ss, payload) {
  const usuarioId = normalizarEmail(payload.usuario_id);
  const dispositivoId = String(payload.dispositivo_id || "").trim();
  const evento = String(payload.evento || "pregunta_libre").trim();
  const modo = String(payload.modo || "auto").toLowerCase() === "chat" ? "chat" : "auto";
  const pregunta = String(payload.pregunta || "").trim();
  const contexto = payload.contexto && typeof payload.contexto === "object" ? payload.contexto : {};
  const esModoTestInterno = Boolean(payload.__test_mode) && String(payload.admin_key || payload.adminKey || "") === ADMIN_KEY;

  if (!coachHabilitadoGlobal()) {
    return finalizarJSON({ status: "disabled", message: "Coach pausado globalmente" });
  }

  if (!esModoTestInterno && (!usuarioId || !dispositivoId || !sesionValidaNutrafit(ss, usuarioId, dispositivoId))) {
    return finalizarJSON({ status: "error", message: "Sesion invalida" });
  }

  const cfg = obtenerConfigCoach();
  const cuota = esModoTestInterno
    ? { ok: true, used: 0, max: 999 }
    : validarCuotaCoachDiaria(ss, usuarioId, modo, cfg);
  if (!cuota.ok) {
    return finalizarJSON({
      status: "quota_exceeded",
      message: "Limite diario del piloto alcanzado",
      quota: cuota
    });
  }

  const modelo = elegirModeloCoach(evento, payload, cfg);
  const accionNutrafit = mapearEventoAAccionNutrafit(evento, payload);
  const intencionDetectada = detectarIntencionCoach(pregunta, evento);
  const rutaRespuesta = clasificarRutaRespuestaCoach(pregunta, evento, modo, intencionDetectada);
  const memoriaUsuario = obtenerMemoriaCoachUsuario(ss, usuarioId, 8);
  const contextoUsuario = construirContextoUsuarioCoach(ss, usuarioId);
  const contextoBaseOficial = obtenerContextoBaseOficialCoach(pregunta, intencionDetectada, rutaRespuesta, 4);
  const contextoAlimento = obtenerContextoAlimentoCoach(ss, pregunta);
  const contextoWeb = preguntaRequiereBusquedaWebNutricion(pregunta, evento, modo, intencionDetectada)
    ? obtenerContextoWebNutricion(pregunta, intencionDetectada)
    : { resumen: "", fuentes: [] };
  const resumenUsuario = construirResumenUsuarioNutrafitParaCoach(contextoUsuario, contexto || {});
  const ejemplosCalibracion = seleccionarEjemplosCalibracionCoach(pregunta, intencionDetectada, rutaRespuesta, 3);
  const contextoFinal = Object.assign({}, contexto || {}, {
    perfil_usuario: contextoUsuario,
    accion_nutrafit: accionNutrafit,
    memoria_usuario: memoriaUsuario,
    intencion_detectada: intencionDetectada,
    ruta_respuesta: rutaRespuesta,
    resumen_usuario_nutrafit: resumenUsuario,
    ejemplos_calibracion: ejemplosCalibracion,
    base_oficial_coach: contextoBaseOficial.items,
    resumen_base_oficial: contextoBaseOficial.resumen,
    alimento_detectado: contextoAlimento.detectado,
    alimento_oficial: contextoAlimento.oficial,
    alimento_catalogo: contextoAlimento.catalogo,
    resumen_alimento: contextoAlimento.resumen,
    resumen_web: contextoWeb.resumen,
    fuentes_web: contextoWeb.fuentes
  });
  const mensajeUsuario = (rutaRespuesta === "factual")
    ? construirMensajeCoachFactual(pregunta, evento, contextoFinal, modo)
    : construirMensajeCoach(pregunta, evento, contextoFinal, modo);
  const promptSistema = rutaRespuesta === "factual" ? promptFactualNutrafit : promptMaestroNutrafit;
  const tempOverride = rutaRespuesta === "factual" ? 0.25 : null;

  try {
    const respuestaGroq = solicitarGroqCoach(modelo, mensajeUsuario, cfg.maxTokensSalida, modo, evento, {
      systemPrompt: promptSistema,
      temperature: tempOverride
    });
    const contenido = String(
      (((respuestaGroq || {}).choices || [])[0] || {}).message
        ? (((respuestaGroq || {}).choices || [])[0] || {}).message.content || ""
        : ""
    ).trim();

    let respuesta = parsearRespuestaCoachJSON(contenido);
    respuesta = normalizarRespuestaCoach(respuesta, evento, contextoFinal, pregunta, contenido, modo);

    if (!esModoTestInterno) {
      registrarMemoriaCoachUsuario(ss, {
        usuarioId: usuarioId,
        pregunta: pregunta,
        respuestaDirecta: String(respuesta.respuesta_directa || "").substring(0, 300),
        accion: accionNutrafit,
        intencion: intencionDetectada
      });
    }

    const usage = respuestaGroq.usage || {};
    if (!esModoTestInterno) {
      registrarUsoCoach(ss, {
        usuarioId,
        modo,
        evento,
        modelo,
        promptTokens: Number(usage.prompt_tokens || 0),
        completionTokens: Number(usage.completion_tokens || 0),
        totalTokens: Number(usage.total_tokens || 0),
        pregunta
      });
    }

    return finalizarJSON({
      status: "ok",
      model: modelo,
      respuesta,
      quota: {
        used: cuota.used + 1,
        max: cuota.max,
        mode: modo
      }
    });
  } catch (error) {
    Logger.log("Error coach_nutrafit: " + error);
    const fallback = normalizarRespuestaCoach({}, evento, contextoFinal, pregunta, "", modo);

    if (!esModoTestInterno) {
      registrarMemoriaCoachUsuario(ss, {
        usuarioId: usuarioId,
        pregunta: pregunta,
        respuestaDirecta: String(fallback.respuesta_directa || "").substring(0, 300),
        accion: accionNutrafit,
        intencion: intencionDetectada
      });

      registrarUsoCoach(ss, {
        usuarioId,
        modo,
        evento,
        modelo,
        promptTokens: 0,
        completionTokens: 0,
        totalTokens: 0,
        pregunta
      });
    }

    return finalizarJSON({
      status: "ok",
      model: modelo,
      respuesta: fallback,
      warning: "fallback"
    });
  }
}

function validarCuotaCoachDiaria(ss, usuarioId, modo, cfg) {
  const hoja = obtenerHojaUsoCoach(ss);
  const fechaHoy = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  const max = modo === "chat" ? cfg.maxChatPorDia : cfg.maxAutoPorDia;

  if (hoja.getLastRow() <= 1) {
    return { ok: true, used: 0, max: max };
  }

  const datos = hoja.getRange(2, 1, hoja.getLastRow() - 1, 10).getValues();
  let usados = 0;
  datos.forEach(fila => {
    const fecha = String(fila[1] || "").trim();
    const uid = normalizarEmail(fila[2] || "");
    const m = String(fila[3] || "").trim();
    if (fecha === fechaHoy && uid === usuarioId && m === modo) {
      usados++;
    }
  });

  return { ok: usados < max, used: usados, max: max };
}

function obtenerMemoriaCoachUsuario(ss, usuarioId, limite) {
  const hoja = obtenerHojaMemoriaCoach(ss);
  if (hoja.getLastRow() <= 1) return [];

  const uid = normalizarEmail(usuarioId);
  const max = Math.max(1, Number(limite || 6));
  const datos = hoja.getRange(2, 1, hoja.getLastRow() - 1, 7).getValues();
  const filtrados = [];

  for (let i = datos.length - 1; i >= 0; i--) {
    const fila = datos[i];
    if (normalizarEmail(fila[2]) !== uid) continue;
    filtrados.push({
      fecha: String(fila[1] || ""),
      accion: String(fila[3] || ""),
      intencion: String(fila[4] || ""),
      pregunta: String(fila[5] || ""),
      respuesta: String(fila[6] || "")
    });
    if (filtrados.length >= max) break;
  }

  return filtrados.reverse();
}

function construirContextoUsuarioCoach(ss, usuarioId) {
  const uid = normalizarEmail(usuarioId);
  const tz = Session.getScriptTimeZone();

  const hoy = Utilities.formatDate(new Date(), tz, "yyyy-MM-dd");
  const resumen = {
    fecha_hoy: hoy,
    agua_hoy_vasos: null,
    creditos_ultimo_total: null,
    creditos_ultimo_fecha: "",
    ejercicio_ultimo: null,
    peso_ultimo: null
  };

  const hojaAgua = ss.getSheetByName("Control_Agua");
  if (hojaAgua && hojaAgua.getLastRow() > 1) {
    const datosAgua = hojaAgua.getDataRange().getValues();
    let vasosHoy = 0;
    for (let i = datosAgua.length - 1; i >= 1; i--) {
      const fila = datosAgua[i];
      if (normalizarEmail(fila[0]) !== uid) continue;
      const fechaFila = String(fila[1] || "").trim();
      if (!(fechaFila === hoy || fechaFila.indexOf(hoy) === 0)) continue;
      const vasosRaw = String(fila[2] || "");
      const m = vasosRaw.match(/(\d+)/);
      if (m) vasosHoy = Math.max(vasosHoy, Number(m[1] || 0));
    }
    resumen.agua_hoy_vasos = vasosHoy;
  }

  const hojaCred = ss.getSheetByName("Creditos") || ss.getSheetByName("CREDITOS");
  if (hojaCred && hojaCred.getLastRow() > 1) {
    const datosCred = hojaCred.getDataRange().getValues();
    for (let i = datosCred.length - 1; i >= 1; i--) {
      const fila = datosCred[i];
      if (normalizarEmail(fila[0]) !== uid) continue;
      resumen.creditos_ultimo_total = Number(fila[6]);
      resumen.creditos_ultimo_fecha = String(fila[1] || "");
      break;
    }
  }

  const hojaEj = ss.getSheetByName("ejercicio");
  if (hojaEj && hojaEj.getLastRow() > 1) {
    const datosEj = hojaEj.getDataRange().getValues();
    for (let i = datosEj.length - 1; i >= 1; i--) {
      const fila = datosEj[i];
      if (normalizarEmail(fila[0]) !== uid) continue;
      resumen.ejercicio_ultimo = {
        fecha: fila[1] instanceof Date ? Utilities.formatDate(fila[1], tz, "yyyy-MM-dd") : String(fila[1] || ""),
        actividad: String(fila[2] || ""),
        tiempo_min: Number(fila[3] || 0),
        distancia: Number(fila[5] || 0),
        pasos: Number(fila[6] || 0)
      };
      break;
    }
  }

  const hojaPeso = ss.getSheetByName("Peso");
  if (hojaPeso && hojaPeso.getLastRow() > 1) {
    const datosPeso = hojaPeso.getDataRange().getValues();
    for (let i = datosPeso.length - 1; i >= 1; i--) {
      const fila = datosPeso[i];
      if (normalizarEmail(fila[0]) !== uid) continue;
      resumen.peso_ultimo = {
        fecha: String(fila[1] || ""),
        peso: Number(fila[2] || 0),
        diferencia: String(fila[3] || "")
      };
      break;
    }
  }

  return resumen;
}

function clasificarRutaRespuestaCoach(pregunta, evento, modo, intencion) {
  const preguntaTxt = String(pregunta || "").trim().toLowerCase();
  const eventoTxt = String(evento || "").trim().toLowerCase();
  const modoTxt = String(modo || "").trim().toLowerCase();
  const intencionTxt = String(intencion || "").trim().toLowerCase();

  if (!(eventoTxt === "pregunta_libre" || modoTxt === "chat")) return "coaching";
  if (intencionTxt === "plan_hoy" || intencionTxt === "motivacion") return "coaching";

  if (/que es|beneficio|beneficios|riesgo|riesgos|caloria|calorias|proteina|hidrato|fibra|vitamina|mineral|omega|indice glucemico|colesterol|trigliceridos|apto|contraindic|preentreno|postentreno|recuperacion|masa muscular|fuerza|resistencia|cardio|series|repeticiones|pasos|hidratacion|electrolitos/.test(preguntaTxt)) {
    return "factual";
  }

  if (["nutricion_general", "comidas", "comer_fuera", "general", "ejercicio", "agua"].indexOf(intencionTxt) >= 0) {
    return "factual";
  }

  return "coaching";
}

function normalizarTextoCoachComparable(texto) {
  return String(texto || "")
    .toLowerCase()
    .replace(/[áàäâ]/g, "a")
    .replace(/[éèëê]/g, "e")
    .replace(/[íìïî]/g, "i")
    .replace(/[óòöô]/g, "o")
    .replace(/[úùüû]/g, "u")
    .replace(/ñ/g, "n")
    .replace(/[^a-z0-9\s]/g, " ")
    .replace(/\s+/g, " ")
    .trim();
}

function tokenizarCoachComparable(texto) {
  const limpio = normalizarTextoCoachComparable(texto);
  if (!limpio) return [];
  return limpio.split(" ").filter(function(token) {
    return token && token.length >= 3;
  });
}

function categoriasPreferidasCoach(intencion, rutaRespuesta) {
  const categorias = MAPA_CATEGORIAS_INTENCION_COACH[String(intencion || "general")] || ["nutricion_general"];
  if (String(rutaRespuesta || "").toLowerCase() === "factual") {
    return categorias.concat(["nutricion_general", "alimentos"]);
  }
  return categorias.concat(["planificacion", "nutrafit_usuario", "comportamiento"]);
}

function seleccionarEjemplosCalibracionCoach(pregunta, intencion, rutaRespuesta, limite) {
  const tokensPregunta = tokenizarCoachComparable(pregunta);
  const preferidas = categoriasPreferidasCoach(intencion, rutaRespuesta);
  const max = Math.max(1, Number(limite || 3));
  const baseCalibracion = COACH_BATERIA_CALIBRACION.concat(
    COACH_MINI_TEST_10.map(function(item) {
      return {
        id: 1000 + Number(item.id || 0),
        categoria: item.categoria,
        pregunta: item.pregunta,
        respuesta: item.respuesta
      };
    })
  );

  const puntuados = baseCalibracion.map(function(item) {
    const tokensItem = tokenizarCoachComparable(item.pregunta + " " + item.respuesta);
    const mapaTokens = {};
    tokensItem.forEach(function(token) { mapaTokens[token] = true; });

    let score = 0;
    if (preferidas.indexOf(String(item.categoria || "")) >= 0) score += 6;

    tokensPregunta.forEach(function(token) {
      if (mapaTokens[token]) score += 2;
    });

    if (esCoincidenciaAltaMiniTestCoach(pregunta, item.pregunta)) {
      score += 20;
    }

    if (String(rutaRespuesta || "").toLowerCase() === "factual" && item.categoria === "alimentos") score += 2;
    if (String(rutaRespuesta || "").toLowerCase() !== "factual" && item.categoria === "planificacion") score += 2;

    return { item: item, score: score };
  });

  return puntuados
    .sort(function(a, b) { return b.score - a.score; })
    .filter(function(entry, idx) {
      return entry.score > 0 || idx < max;
    })
    .slice(0, max)
    .map(function(entry) {
      return {
        id: entry.item.id,
        categoria: entry.item.categoria,
        pregunta: entry.item.pregunta,
        respuesta_esperada: entry.item.respuesta
      };
    });
}

function formatearEjemplosCalibracionCoach(ejemplos) {
  const lista = Array.isArray(ejemplos) ? ejemplos : [];
  if (!lista.length) return "Sin ejemplos de calibracion.";

  return lista.map(function(item) {
    return [
      "- Categoria: " + String(item.categoria || "general"),
      "  Pregunta modelo: " + String(item.pregunta || ""),
      "  Respuesta esperada modelo: " + String(item.respuesta_esperada || "")
    ].join("\n");
  }).join("\n");
}

function puntuarBaseOficialCoach(item, tokensPregunta, intencion, rutaRespuesta) {
  let score = 0;
  const cats = Array.isArray(item.categorias) ? item.categorias : [];
  const tags = Array.isArray(item.etiquetas) ? item.etiquetas : [];
  const contenido = String(item.contenido || "");
  const tokenSet = {};

  tokenizarCoachComparable(tags.join(" ") + " " + contenido).forEach(function(token) {
    tokenSet[token] = true;
  });

  if (cats.indexOf(String(intencion || "general")) >= 0) score += 8;
  if (String(rutaRespuesta || "").toLowerCase() === "factual" && (cats.indexOf("nutricion_general") >= 0 || cats.indexOf("ejercicio") >= 0 || cats.indexOf("alimentos") >= 0)) {
    score += 2;
  }

  tokensPregunta.forEach(function(token) {
    if (tokenSet[token]) score += 3;
  });

  return score;
}

function obtenerContextoBaseOficialCoach(pregunta, intencion, rutaRespuesta, limite) {
  const tokensPregunta = tokenizarCoachComparable(pregunta);
  const max = Math.max(1, Number(limite || 4));
  const items = BASE_OFICIAL_COACH_NUTRAFIT
    .map(function(item) {
      return {
        item: item,
        score: puntuarBaseOficialCoach(item, tokensPregunta, intencion, rutaRespuesta)
      };
    })
    .sort(function(a, b) { return b.score - a.score; })
    .filter(function(entry, idx) {
      return entry.score > 0 || idx < max;
    })
    .slice(0, max)
    .map(function(entry) {
      return {
        id: entry.item.id,
        categorias: entry.item.categorias,
        etiquetas: entry.item.etiquetas,
        contenido: entry.item.contenido,
        fuentes: entry.item.fuentes
      };
    });

  const resumen = items.length
    ? ("Base oficial interna encontrada: " + items.length + " ficha(s) curadas para esta consulta.")
    : "Sin fichas internas relevantes para esta consulta.";

  return { resumen: resumen, items: items };
}

function obtenerRegistrosAlimentosNutrafit(ss) {
  const hoja = obtenerHojaAlimentosNutrafit(ss);
  if (!hoja || hoja.getLastRow() <= 1) return [];

  const data = hoja.getDataRange().getValues();
  const headersRaw = data[0].map(function(v) { return String(v || "").trim(); });
  const headersNorm = headersRaw.map(function(v) { return normalizarClaveAlimentoCoach(v); });
  const body = data.slice(1);

  function idx() {
    for (var a = 0; a < arguments.length; a++) {
      var pos = headersNorm.indexOf(normalizarClaveAlimentoCoach(arguments[a]));
      if (pos >= 0) return pos;
    }
    return -1;
  }

  var idxNombre = idx("nombre", "alimento");
  var idxGrupo = idx("grupo", "categoria");
  var idxProt = idx("proteinas", "proteina");
  var idxCarb = idx("carbohidratos", "hidratos");
  var idxGrasa = idx("grasas", "grasa");
  var idxFibra = idx("fibra");
  var idxManual = idx("manual");
  var idxCalculado = idx("calculado");
  var idxNeto = idx("neto", "netos", "puntos", "creditos");

  if (idxNombre === -1) idxNombre = 0;

  return body.map(function(fila) {
    var nombre = String(fila[idxNombre] || "").trim();
    if (!nombre) return null;
    return {
      nombre: nombre,
      clave: normalizarClaveAlimentoCoach(nombre),
      grupo: idxGrupo >= 0 ? String(fila[idxGrupo] || "").trim() : "",
      proteinas: idxProt >= 0 ? Number(fila[idxProt] || 0) : null,
      carbohidratos: idxCarb >= 0 ? Number(fila[idxCarb] || 0) : null,
      grasas: idxGrasa >= 0 ? Number(fila[idxGrasa] || 0) : null,
      fibra: idxFibra >= 0 ? Number(fila[idxFibra] || 0) : null,
      manual: idxManual >= 0 ? Number(fila[idxManual] || 0) : null,
      calculado: idxCalculado >= 0 ? Number(fila[idxCalculado] || 0) : null,
      neto: idxNeto >= 0 ? Number(fila[idxNeto] || 0) : null
    };
  }).filter(function(item) { return item !== null; });
}

function buscarAlimentoCatalogoCoach(nombreDetectado, catalogo) {
  var clave = normalizarClaveAlimentoCoach(nombreDetectado);
  if (!clave) return null;

  var mejor = null;
  var mejorLongitud = -1;
  (Array.isArray(catalogo) ? catalogo : CATALOGO_ALIMENTOS_COACH).forEach(function(item) {
    var aliases = Array.isArray(item.aliases) ? item.aliases : [item.nombre];
    aliases.forEach(function(alias) {
      var aliasClave = normalizarClaveAlimentoCoach(alias);
      if (!aliasClave) return;
      if (clave === aliasClave || clave.indexOf(aliasClave) >= 0 || aliasClave.indexOf(clave) >= 0) {
        if (aliasClave.length > mejorLongitud) {
          mejor = item;
          mejorLongitud = aliasClave.length;
        }
      }
    });
  });

  return mejor;
}

function detectarAlimentoPreguntaCoach(ss, pregunta) {
  var texto = normalizarClaveAlimentoCoach(pregunta);
  if (!texto) return { detectado: "", oficial: null, catalogo: null };
  var catalogoCoach = obtenerCatalogoAlimentosCoach(ss);

  var mejorNombre = "";
  var mejorLongitud = -1;
  var mejorCatalogo = null;

  catalogoCoach.forEach(function(item) {
    var aliases = Array.isArray(item.aliases) ? item.aliases : [item.nombre];
    aliases.forEach(function(alias) {
      var aliasClave = normalizarClaveAlimentoCoach(alias);
      if (!aliasClave) return;
      if (texto.indexOf(aliasClave) >= 0 && aliasClave.length > mejorLongitud) {
        mejorNombre = item.nombre;
        mejorLongitud = aliasClave.length;
        mejorCatalogo = item;
      }
    });
  });

  var mejorOficial = null;
  obtenerRegistrosAlimentosNutrafit(ss).forEach(function(item) {
    if (item.clave && texto.indexOf(item.clave) >= 0 && item.clave.length > mejorLongitud) {
      mejorNombre = item.nombre;
      mejorLongitud = item.clave.length;
      mejorOficial = item;
    }
  });

  if (!mejorCatalogo && mejorNombre) {
    mejorCatalogo = buscarAlimentoCatalogoCoach(mejorNombre, catalogoCoach);
  }

  return {
    detectado: mejorNombre,
    oficial: mejorOficial,
    catalogo: mejorCatalogo
  };
}

function solicitarGroqCoach(modelo, mensajeUsuario, maxTokensSalida, modo, evento, opciones) {
  const props = PropertiesService.getScriptProperties();
  const apiKey = props.getProperty("GROQ_API_KEY");
  if (!apiKey) {
    throw new Error("Falta GROQ_API_KEY en Script Properties");
  }

  const modoTxt = String(modo || "auto").toLowerCase() === "chat" ? "chat" : "auto";
  const eventoTxt = String(evento || "").toLowerCase();
  const esChatLibre = (modoTxt === "chat" || eventoTxt === "pregunta_libre");
  const temperaturaBase = esChatLibre ? 0.9 : 0.5;
  const temperature = (opciones && typeof opciones.temperature === "number") ? Number(opciones.temperature) : temperaturaBase;
  const maxTokensFinal = Math.max(Number(maxTokensSalida || 260), modoTxt === "chat" ? 320 : 260);
  const systemPrompt = String((opciones && opciones.systemPrompt) || promptMaestroNutrafit);

  const payload = {
    model: modelo,
    temperature: temperature,
    max_tokens: maxTokensFinal,
    messages: [
      { role: "system", content: systemPrompt },
      { role: "user", content: mensajeUsuario }
    ],
    response_format: { type: "json_object" }
  };

  const response = UrlFetchApp.fetch("https://api.groq.com/openai/v1/chat/completions", {
    method: "post",
    contentType: "application/json",
    headers: {
      Authorization: "Bearer " + apiKey
    },
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  });

  const status = response.getResponseCode();
  const body = response.getContentText();
  if (status < 200 || status >= 300) {
    throw new Error("Groq HTTP " + status + " - " + body.substring(0, 500));
  }

  return JSON.parse(body || "{}");
}

function parsearRespuestaCoachJSON(texto) {
  if (!texto) return {};

  try {
    return JSON.parse(texto);
  } catch (_) {
    const match = String(texto).match(/\{[\s\S]*\}/);
    if (!match) return {};
    try {
      return JSON.parse(match[0]);
    } catch (_) {
      const textoPlano = String(texto || "").trim();
      return textoPlano ? { __textoPlano: textoPlano } : {};
    }
  }
}

function calcularSolapeTokensCoach(textoA, textoB) {
  const a = tokenizarCoachComparable(textoA || "");
  const b = tokenizarCoachComparable(textoB || "");
  if (!a.length || !b.length) return 0;

  const mapA = {};
  const mapB = {};
  a.forEach(function(t) { mapA[t] = true; });
  b.forEach(function(t) { mapB[t] = true; });

  let inter = 0;
  Object.keys(mapA).forEach(function(k) {
    if (mapB[k]) inter++;
  });

  const union = Object.keys(mapA).length + Object.keys(mapB).length - inter;
  if (!union) return 0;
  return inter / union;
}

function esCoincidenciaAltaMiniTestCoach(preguntaA, preguntaB) {
  const pa = normalizarTextoCoachComparable(preguntaA || "");
  const pb = normalizarTextoCoachComparable(preguntaB || "");
  if (!pa || !pb) return false;
  if (pa === pb) return true;
  if (pa.indexOf(pb) >= 0 || pb.indexOf(pa) >= 0) return true;
  return calcularSolapeTokensCoach(pa, pb) >= 0.62;
}

function obtenerRespuestaMiniTestCoach(preguntaTxt) {
  const preguntaNorm = normalizarTextoCoachComparable(preguntaTxt || "");
  if (!preguntaNorm) return null;

  let mejor = null;
  COACH_MINI_TEST_10.forEach(function(item) {
    const score = esCoincidenciaAltaMiniTestCoach(preguntaNorm, item.pregunta)
      ? 1
      : calcularSolapeTokensCoach(preguntaNorm, item.pregunta);
    if (!mejor || score > mejor.score) {
      mejor = { score: score, item: item };
    }
  });

  if (!mejor || mejor.score < 0.62) return null;
  return mejor.item;
}

  const disponibles = Number(contexto.creditos_disponibles);
  const consumidos = Number(contexto.creditos_consumidos);
  const tieneCreditos = !isNaN(disponibles) && !isNaN(consumidos);
  const exceso = tieneCreditos && disponibles < 0 ? Math.abs(disponibles) : 0;

  const estado = String(respuesta.estado_dia || "").toUpperCase();
  const estadoFinal = (estado === "VERDE" || estado === "AMARILLO" || estado === "ROJO")
    ? estado
    : (exceso > 0 ? "ROJO" : (tieneCreditos && disponibles <= 3 ? "AMARILLO" : "VERDE"));

  const preguntaTxt = String(pregunta || "").trim();
  const eventoTxt = String(evento || "").trim().toLowerCase();
  const textoPlano = String(respuesta.__textoPlano || "").trim();
  const contenidoPlano = String(contenidoRaw || "").trim();
  const intencion = detectarIntencionCoach(preguntaTxt, eventoTxt);
  const modoTxt = String(modo || "auto").toLowerCase() === "chat" ? "chat" : "auto";
  const esPreguntaDirecta = (modoTxt === "chat" || eventoTxt === "pregunta_libre");
  const rutaRespuesta = String(contexto.ruta_respuesta || "coaching").toLowerCase();
  const esRutaFactual = rutaRespuesta === "factual";
  const plantilla = construirFallbackSegunIntencion(intencion, {
    exceso: exceso,
    disponibles: disponibles,
    consumidos: consumidos,
    pregunta: preguntaTxt
  });

  const lecturaDinamica = tieneCreditos
    ? (exceso > 0
      ? "Hay un exceso en el consumo de creditos. Te ayudo a compensarlo con decisiones concretas."
      : (disponibles <= 3
        ? "Vas justa de creditos y estas a tiempo de cerrar bien el dia."
        : "Vas con buen margen para mantener equilibrio."))
    : "Reviso tu situacion y te doy el siguiente paso mas util ahora.";

  const lecturaPregunta = preguntaTxt
    ? "Sobre tu pregunta: " + preguntaTxt
    : "";

  const lecturaBase = respuesta.lectura_rapida
    ? String(respuesta.lectura_rapida)
    : (textoPlano || contenidoPlano || (lecturaPregunta ? (lecturaPregunta + ". " + lecturaDinamica) : lecturaDinamica));

  const respuestaDirectaRaw = String(respuesta.respuesta_directa || textoPlano || "").trim();
  let respuestaDirecta = limpiarRespuestaDirectaCoach(respuestaDirectaRaw);

  if (esRutaFactual) {
    const fuentes = Array.isArray(contexto.fuentes_web) ? contexto.fuentes_web.filter(Boolean).slice(0, 3) : [];
    const tieneDatosOficiales = contextoTieneDatosOficialesNutricion(contexto);
    const esConsultaMedica = esConsultaMedicaNutricion(preguntaTxt);
    const respuestaCatalogo = construirRespuestaDirectaDesdeCatalogoAlimento(contexto, preguntaTxt);
    const respuestaChiaCanonica = construirRespuestaCanonicaChiaCoach(preguntaTxt);
    const respuestaMiniTest = obtenerRespuestaMiniTestCoach(preguntaTxt);

    if (esConsultaMedica) {
      return {
        respuesta_directa: "Soy un coach nutricional, no un profesional medico.",
        estado_dia: "",
        lectura_rapida: "",
        accion_ahora: "",
        siguiente_paso: "",
        ajuste_creditos: "",
        micro_habito: "",
        mensaje_motivador: ""
      };
    }

    const esPreguntaAlimentoONutricion = /alimento|comida|nutricion|nutricional|caloria|calorias|proteina|grasa|hidrato|fibra|vitamina|mineral|omega|azucar|porcion|ingrediente/.test(preguntaTxt.toLowerCase());
    if (esPreguntaAlimentoONutricion && !tieneDatosOficiales && fuentes.length === 0 && !respuestaCatalogo) {
      return {
        respuesta_directa: "Necesito datos nutricionales oficiales para darte una respuesta exacta. Enviame la informacion del alimento.",
        estado_dia: "",
        lectura_rapida: "",
        accion_ahora: "",
        siguiente_paso: "",
        ajuste_creditos: "",
        micro_habito: "",
        mensaje_motivador: ""
      };
    }

    const baseFactual = respuestaDirecta
      || respuestaCatalogo
      || limpiarRespuestaDirectaCoach(textoPlano)
      || (preguntaTxt ? ("No tengo evidencia suficiente para responder con precision a: " + preguntaTxt + ". Dame mas contexto (cantidad, frecuencia y objetivo).") : "Necesito tu pregunta concreta para ayudarte mejor.");

    const respuestaFactualFinal = respuestaChiaCanonica
      || (respuestaMiniTest ? String(respuestaMiniTest.respuesta || "") : "")
      || baseFactual;

    return {
      respuesta_directa: limpiarRespuestaDirectaCoach(respuestaFactualFinal),
      estado_dia: "",
      lectura_rapida: "",
      accion_ahora: "",
      siguiente_paso: "",
      ajuste_creditos: "",
      micro_habito: "",
      mensaje_motivador: ""
    };
  }

  if (intencion === "plan_hoy" && (!respuestaDirecta || esPlantillaAntiguaCoach(respuestaDirecta) || respuestaDirecta.length < 90)) {
    respuestaDirecta = construirPlanHoyDirectoCoach(contexto);
  }

  if (!respuestaDirecta && esPreguntaDirecta) {
    const fallbackDirecto = [plantilla.accion_ahora, plantilla.siguiente_paso].filter(Boolean).join(" ").trim();
    respuestaDirecta = limpiarRespuestaDirectaCoach(fallbackDirecto);
  }

  if (esPreguntaDirecta) {
    return {
      respuesta_directa: respuestaDirecta || (preguntaTxt ? ("Necesito un dato mas para responder con precision a tu pregunta: " + preguntaTxt) : "Necesito tu pregunta concreta para ayudarte mejor."),
      estado_dia: "",
      lectura_rapida: "",
      accion_ahora: "",
      siguiente_paso: "",
      ajuste_creditos: "",
      micro_habito: "",
      mensaje_motivador: ""
    };
  }

  let lecturaFinal = lecturaBase;
  if (exceso > 0) {
    const objetivo = Math.max(0, consumidos - exceso);
    lecturaFinal = "Hay un exceso en el consumo de creditos. Objetivo: " + objetivo + ", consumidos: " + consumidos + ", exceso: " + exceso + ". " + lecturaBase;
  }

  let siguientePaso = String(respuesta.siguiente_paso || plantilla.siguiente_paso);
  if (exceso > 0) {
    siguientePaso = "Para el siguiente paso, usa verduras cocinadas y una proteina magra. Deja crudos solo para la cena.";
  }

  const accionAhoraDefault = plantilla.accion_ahora;
  const siguientePasoDefault = plantilla.siguiente_paso;
  const microHabitoDefault = plantilla.micro_habito;
  const motivadorDefault = plantilla.mensaje_motivador;

  return {
    respuesta_directa: respuestaDirecta,
    estado_dia: estadoFinal,
    lectura_rapida: lecturaFinal,
    accion_ahora: String(respuesta.accion_ahora || accionAhoraDefault),
    siguiente_paso: String(respuesta.siguiente_paso || siguientePaso || siguientePasoDefault),
    ajuste_creditos: String(respuesta.ajuste_creditos || (exceso > 0
      ? "Compensa con ejercicio moderado hoy o manana y ajusta una comida, sin castigos."
      : "Mantiene el reparto equilibrado de creditos para llegar con margen a la noche.")),
    micro_habito: String(respuesta.micro_habito || microHabitoDefault),
    mensaje_motivador: String(respuesta.mensaje_motivador || motivadorDefault)
  };
}

/**
 * Sanitiza valores antes de guardar
 */
function sanitizeCampo(valor) {
  if (valor === null || valor === undefined) return "";
  if (typeof valor === "string") return valor.trim();
  return valor;
}

function esConsultaMedicaNutricion(preguntaTxt) {
  const t = String(preguntaTxt || "").toLowerCase();
  return /diagnost|tratamiento|medicacion|medicamento|enfermedad|patologia|hipertension|diabetes|cancer|tiroides|embarazo de riesgo|insuficiencia|renal|hepatica/.test(t);
}

function contextoTieneDatosOficialesNutricion(contexto) {
  const c = contexto && typeof contexto === "object" ? contexto : {};

  const candidatos = [
    c.datos_nutricionales,
    c.nutricion,
    c.alimento,
    c.alimento_info,
    c.fooddata,
    c.foodDataCentral,
    c.tabla_nutricional,
    c.porciones,
    c.nutrientes,
    c.macros,
    c.alimento_oficial,
    c.alimento_encontrado,
    c.resumen_alimento
  ];

  for (let i = 0; i < candidatos.length; i++) {
    const v = candidatos[i];
    if (!v) continue;
    if (typeof v === "string" && v.trim().length > 0) return true;
    if (typeof v === "number" && !isNaN(v)) return true;
    if (Array.isArray(v) && v.length > 0) return true;
    if (typeof v === "object" && Object.keys(v).length > 0) return true;
  }

  const creditosDisponibles = Number(c.creditos_disponibles);
  const creditosConsumidos = Number(c.creditos_consumidos);
  if (!isNaN(creditosDisponibles) || !isNaN(creditosConsumidos)) return true;

  const perfil = c.perfil_usuario && typeof c.perfil_usuario === "object" ? c.perfil_usuario : null;
  if (perfil) {
    if (perfil.creditos_ultimo_total !== null && perfil.creditos_ultimo_total !== undefined) return true;
    if (perfil.agua_hoy_vasos !== null && perfil.agua_hoy_vasos !== undefined) return true;
    if (perfil.peso_ultimo || perfil.ejercicio_ultimo) return true;
  }

  return false;
}

function evaluarCasoMiniTestCoach(caso, respuestaDirecta) {
  const texto = String(respuestaDirecta || "").trim();
  const textoNorm = normalizarTextoCoachComparable(texto);
  const claves = Array.isArray(caso.claves_minimas) ? caso.claves_minimas : [];

  const contieneFuentes = /fuentes?\s*(consultadas?|usadas?)|https?:\/\//i.test(texto);
  const faltantes = claves.filter(function(k) {
    const key = normalizarTextoCoachComparable(k || "");
    return key && textoNorm.indexOf(key) === -1;
  });

  const aprobado = Boolean(texto) && !contieneFuentes && faltantes.length <= 1;

  return {
    aprobado: aprobado,
    contiene_fuentes: contieneFuentes,
    claves_faltantes: faltantes,
    longitud: texto.length
  };
}

function ejecutarMiniTestCoach(ss, payload) {
  const adminKey = String(payload.admin_key || payload.adminKey || "").trim();
  const usuarioId = normalizarEmail(payload.usuario_id);
  const dispositivoId = String(payload.dispositivo_id || "").trim();
  const autorizadoPorSesion = Boolean(usuarioId && dispositivoId && sesionValidaNutrafit(ss, usuarioId, dispositivoId));

  if (!(adminKey === ADMIN_KEY || autorizadoPorSesion)) {
    return finalizarJSON({ status: "error", message: "No autorizado para ejecutar mini test" });
  }

  const resultados = COACH_MINI_TEST_10.map(function(caso) {
    const payloadCaso = {
      tipo: "coach_nutrafit",
      usuario_id: usuarioId,
      dispositivo_id: dispositivoId,
      evento: "pregunta_libre",
      modo: "chat",
      pregunta: caso.pregunta,
      contexto: {},
      __test_mode: true,
      admin_key: adminKey
    };

    let respuestaDirecta = "";
    let status = "error";
    try {
      const salida = procesarCoachNutrafit(ss, payloadCaso);
      const json = JSON.parse(String(salida.getContent() || "{}"));
      status = String(json.status || "error");
      respuestaDirecta = String((((json || {}).respuesta || {}).respuesta_directa) || "");
    } catch (_) {
      status = "error";
      respuestaDirecta = "";
    }

    const evaluacion = evaluarCasoMiniTestCoach(caso, respuestaDirecta);

    return {
      id: caso.id,
      categoria: caso.categoria,
      pregunta: caso.pregunta,
      status: status,
      aprobado: evaluacion.aprobado,
      contiene_fuentes: evaluacion.contiene_fuentes,
      claves_faltantes: evaluacion.claves_faltantes,
      respuesta_esperada: caso.respuesta,
      respuesta_obtenida: respuestaDirecta
    };
  });

  const aprobados = resultados.filter(function(r) { return r.aprobado; }).length;
  const total = resultados.length;
  const porcentaje = total ? Math.round((aprobados / total) * 100) : 0;

  return finalizarJSON({
    status: "ok",
    total: total,
    aprobados: aprobados,
    fallidos: Math.max(0, total - aprobados),
    porcentaje_aprobado: porcentaje,
    resultados: resultados
  });
}

function preguntaRequiereBusquedaWebNutricion(pregunta, evento, modo, intencion) {
  const preguntaTxt = String(pregunta || "").trim();
  const preguntaLower = preguntaTxt.toLowerCase();
  const eventoTxt = String(evento || "").toLowerCase();
  const modoTxt = String(modo || "").toLowerCase();
  const intencionTxt = String(intencion || "");

  if (!preguntaTxt) return false;

  const esChatPregunta = (eventoTxt === "pregunta_libre") || (modoTxt === "chat");
  if (!esChatPregunta) return false;

  const esPreguntaPersonalNutrafit = /\b(mi|mis)\b/.test(preguntaLower)
    && /creditos|agua|peso|progreso|historial|registro|diario/.test(preguntaLower);
  if (esPreguntaPersonalNutrafit) return false;

  if (intencionTxt === "plan_hoy") return false;

  if (["nutricion_general", "comidas", "comer_fuera", "creditos", "general", "ejercicio", "agua"].indexOf(intencionTxt) >= 0) {
    return true;
  }

  return /alimento|comida|nutricion|nutricional|caloria|calorias|proteina|grasa|hidrato|fibra|vitamina|mineral|omega|azucar|salud|deporte|deportivo|entreno|entrenamiento|gym|gimnasio|correr|carrera|fuerza|musculo|masa muscular|recuperacion|hidratacion|electrolitos|cardio|resistencia/.test(preguntaLower);
}

function construirConsultaWebCoach(consulta, intencion) {
  const base = String(consulta || "").trim();
  const intent = String(intencion || "general").toLowerCase();
  if (!base) return "";

  if (intent === "ejercicio") {
    return base + " ejercicio rendimiento deportivo actividad fisica";
  }

  if (intent === "agua") {
    return base + " hidratacion ejercicio salud";
  }

  if (intent === "comer_fuera") {
    return base + " alimentacion saludable nutricion";
  }

  return base + " nutricion alimentacion salud";
}

function dominioPermitidoNutricion(url) {
  const texto = String(url || "").toLowerCase();
  if (!texto) return false;
  for (let i = 0; i < FUENTES_NUTRICION_CONFIABLES.length; i++) {
    const dominio = FUENTES_NUTRICION_CONFIABLES[i].dominio;
    if (texto.indexOf(dominio) >= 0) return true;
  }
  return false;
}

function limpiarHtmlATextoNutricion(html) {
  return String(html || "")
    .replace(/<script[\s\S]*?<\/script>/gi, " ")
    .replace(/<style[\s\S]*?<\/style>/gi, " ")
    .replace(/<[^>]+>/g, " ")
    .replace(/&nbsp;|&#160;/gi, " ")
    .replace(/&amp;/gi, "&")
    .replace(/&quot;/gi, "\"")
    .replace(/\s+/g, " ")
    .trim();
}

function extraerDescripcionDesdeHtmlNutricion(html) {
  const raw = String(html || "");
  const m1 = raw.match(/<meta[^>]+name=["']description["'][^>]+content=["']([^"']+)["'][^>]*>/i);
  if (m1 && m1[1]) return limpiarHtmlATextoNutricion(m1[1]);
  const m2 = raw.match(/<meta[^>]+property=["']og:description["'][^>]+content=["']([^"']+)["'][^>]*>/i);
  if (m2 && m2[1]) return limpiarHtmlATextoNutricion(m2[1]);

  const txt = limpiarHtmlATextoNutricion(raw);
  if (txt.length > 260) return txt.substring(0, 260).trim() + "...";
  return txt;
}

function extraerEnlacesConfiablesDuckDuckGo(consulta, maxResultados, intencion) {
  const consultaBase = String(consulta || "").trim();
  if (!consultaBase) return [];

  const dominios = FUENTES_NUTRICION_CONFIABLES.map(f => "site:" + f.dominio).join(" OR ");
  const q = construirConsultaWebCoach(consultaBase, intencion) + " " + dominios;
  const url = "https://duckduckgo.com/html/?q=" + encodeURIComponent(q);
  const enlaces = [];
  const vistos = {};

  try {
    const resp = UrlFetchApp.fetch(url, { method: "get", muteHttpExceptions: true });
    if (resp.getResponseCode() < 200 || resp.getResponseCode() >= 300) return [];

    const html = resp.getContentText() || "";
    const rgx = /<a[^>]+class=["'][^"']*result__a[^"']*["'][^>]+href=["']([^"']+)["'][^>]*>/gi;
    let m;
    while ((m = rgx.exec(html)) !== null && enlaces.length < Math.max(1, Number(maxResultados || 3))) {
      let href = String(m[1] || "").trim();
      if (!href) continue;

      const uddg = href.match(/[?&]uddg=([^&]+)/i);
      if (uddg && uddg[1]) href = decodeURIComponent(uddg[1]);
      if (!/^https?:\/\//i.test(href)) continue;
      if (!dominioPermitidoNutricion(href)) continue;

      const clave = href.toLowerCase();
      if (vistos[clave]) continue;
      vistos[clave] = true;
      enlaces.push(href);
    }
  } catch (_) {}

  return enlaces;
}

function obtenerContextoWebNutricion(pregunta, intencion) {
  const consulta = String(pregunta || "").trim();
  if (!consulta) return { resumen: "", fuentes: [] };
  const queryCoach = construirConsultaWebCoach(consulta, intencion);

  const fuentes = [];
  const vistos = {};

  function agregarFuente(texto, url) {
    const limpio = String(texto || "").replace(/\s+/g, " ").trim();
    const link = String(url || "").trim();
    if (!limpio || limpio.length < 40) return;

    const clave = limpio.toLowerCase();
    if (vistos[clave]) return;
    vistos[clave] = true;

    const item = link ? (limpio + " (" + link + ")") : limpio;
    fuentes.push(item);
  }

  try {
    const urlWiki = "https://es.wikipedia.org/w/api.php?action=opensearch&search=" + encodeURIComponent(queryCoach) + "&limit=3&namespace=0&format=json";
    const respWiki = UrlFetchApp.fetch(urlWiki, {
      method: "get",
      muteHttpExceptions: true
    });

    if (respWiki.getResponseCode() >= 200 && respWiki.getResponseCode() < 300) {
      const dataWiki = JSON.parse(respWiki.getContentText() || "[]");
      const titulos = Array.isArray(dataWiki[1]) ? dataWiki[1] : [];
      const links = Array.isArray(dataWiki[3]) ? dataWiki[3] : [];

      for (let i = 0; i < titulos.length && fuentes.length < 4; i++) {
        const titulo = String(titulos[i] || "").trim();
        const link = String(links[i] || "").trim();
        if (!titulo) continue;

        try {
          const urlSummary = "https://es.wikipedia.org/api/rest_v1/page/summary/" + encodeURIComponent(titulo);
          const respSummary = UrlFetchApp.fetch(urlSummary, {
            method: "get",
            muteHttpExceptions: true
          });

          if (respSummary.getResponseCode() >= 200 && respSummary.getResponseCode() < 300) {
            const dataSummary = JSON.parse(respSummary.getContentText() || "{}");
            agregarFuente(dataSummary.extract, dataSummary.content_urls && dataSummary.content_urls.desktop ? dataSummary.content_urls.desktop.page : link);
          }
        } catch (_) {
          continue;
        }
      }
    }
  } catch (_) {}

  try {
    const enlaces = extraerEnlacesConfiablesDuckDuckGo(consulta, 4, intencion);
    for (let i = 0; i < enlaces.length && fuentes.length < 6; i++) {
      const enlace = enlaces[i];
      try {
        const resp = UrlFetchApp.fetch(enlace, {
          method: "get",
          muteHttpExceptions: true,
          followRedirects: true
        });
        if (resp.getResponseCode() < 200 || resp.getResponseCode() >= 300) continue;
        const descripcion = extraerDescripcionDesdeHtmlNutricion(resp.getContentText() || "");
        agregarFuente(descripcion, enlace);
      } catch (_) {
        continue;
      }
    }
  } catch (_) {}

  const fuentesFinal = fuentes.slice(0, 4);
  const resumen = fuentesFinal.length
    ? ("Contexto web encontrado en fuentes nutricionales confiables: " + fuentesFinal.length + " fuente(s). Usa consenso general y evita afirmaciones medicas tajantes.")
    : "Sin fuentes web disponibles en este momento. Responde con prudencia y pide precision si falta contexto.";

  return { resumen: resumen, fuentes: fuentesFinal };
}
